// Some earlier bakes used hyphen-minus " - " instead of en dash " – " in Drive
// object names. Status lookups must tolerate both without creating duplicates.

function nameVariants_(s) {
  // Return a small set of likely name variants (no exhaustive normalization).
  if (!s) return [''];
  var out = {};
  function add_(v) { if (v != null && v !== '') out[String(v)] = true; }

  add_(s);
  // Swap spaced separators.
  add_(String(s).split(' – ').join(' - '));
  add_(String(s).split(' - ').join(' – '));
  // Swap raw characters too, in case names were built without spaces.
  add_(String(s).split('–').join('-'));
  add_(String(s).split('-').join('–'));

  return Object.keys(out);
}

function firstFileByNameVariants_(folder, names) {
  if (!folder) return null;
  for (var i = 0; i < names.length; i++) {
    try {
      var it = folder.getFilesByName(names[i]);
      if (it.hasNext()) return it.next();
    } catch (e) {}
  }
  return null;
}



function moveFileToFolder_(file, folder) {
  if (!file || !folder) return;
  try {
    var folderId = folder.getId && folder.getId();

    // Helper: check if file currently has folder as a parent.
    function hasParent_() {
      try {
        var parents = file.getParents();
        while (parents.hasNext()) {
          var p = parents.next();
          if (p && p.getId && p.getId() === folderId) return true;
        }
      } catch (_) {}
      return false;
    }

    // 1) Add to target folder if needed.
    if (!hasParent_()) {
      try { rcFolderAddFile_(folder, file); } catch (_) {}
    }

    // 2) Remove from other parents (best effort) to keep Drive tidy.
    try {
      var ps = file.getParents();
      while (ps.hasNext()) {
        var p2 = ps.next();
        if (p2 && p2.getId && p2.getId() !== folderId) {
          try { p2.removeFile(file); } catch (_) {}
        }
      }
    } catch (_) {}

    // 3) Settle delay + verify (Drive indexing can lag). Retry once if needed.
    try { Utilities.sleep(600); } catch (_) {}
    if (!hasParent_()) {
      try { rcFolderAddFile_(folder, file); } catch (_) {}
      try { Utilities.sleep(600); } catch (_) {}

      // Re-run parent cleanup once after retry.
      try {
        var ps2 = file.getParents();
        while (ps2.hasNext()) {
          var p3 = ps2.next();
          if (p3 && p3.getId && p3.getId() !== folderId) {
            try { p3.removeFile(file); } catch (_) {}
          }
        }
      } catch (_) {}
    }
  } catch (_) {}
}


function ensureEntriesResponsesSpreadsheet_(rallyName, rallyFolder) {
  var name = rallyName || 'Rally';
  // Entries Responses live inside the rally folder (same folder as the working spreadsheet),
  // but as a separate file to keep the working workbook clean.
  var folder = rallyFolder || ensureRallyStorage_(name).rallyFolder;
  var base = 'Entries Responses – ' + name;
  var variants = nameVariants_(base);

  // Try find existing by name variants.
  var f = firstFileByNameVariants_(folder, variants);
  if (f) {
    try { return SpreadsheetApp.openById(f.getId()); } catch (_) {}
  }

  // Create new, then move into the base folder.
  var ss = rcSpreadsheetCreate_(base);
  try {
    var file = rcDriveGetFileById_(ss.getId());
    moveFileToFolder_(file, folder);
  } catch (_) {}

  return ss;
}

/**
 * v2.93 Drive retry hardening.
 * Google Drive/Form services can intermittently throw "Service error: Drive"
 * during create/open/move operations. Retry a few times before failing.
 */
function rcRetry_(label, fn, attempts, delayMs) {
  attempts = attempts || 4;
  delayMs = delayMs || 700;
  var lastErr = null;
  for (var i = 0; i < attempts; i++) {
    try {
      return fn();
    } catch (err) {
      lastErr = err;
      try { Utilities.sleep(delayMs * (i + 1)); } catch (_) {}
    }
  }
  throw lastErr;
}

// === Google Form rally-day windows (03:00 boundary) ==========================
const FORM_DAY_SWITCH_HOUR = 3;
const FORM_WINDOW_TRIGGER_PREFIX = 'FORM_WINDOW_TRIGGER_';

/**
 * Parse the rally date supplied by Controls. Preferred format is YYYY-MM-DD,
 * but DD-MM-YYYY / DD/MM/YYYY and normal Date values are also accepted.
 * Returns a local-midnight Date, or null if the value cannot be resolved.
 */
function parseRallyDateLabel_(value) {
  if (value == null || value === '') return null;

  if (Object.prototype.toString.call(value) === '[object Date]') {
    if (isNaN(value.getTime())) return null;
    return new Date(value.getFullYear(), value.getMonth(), value.getDate(), 0, 0, 0, 0);
  }

  var s = String(value).trim();
  if (!s) return null;
  var m;

  // ISO / year-first: 2026-08-10 or 2026/08/10
  m = /^(\d{4})[-\/](\d{1,2})[-\/](\d{1,2})$/.exec(s);
  if (m) return makeLocalDate_(Number(m[1]), Number(m[2]), Number(m[3]));

  // Rally display form: 10-08-2026 or 10/08/2026
  m = /^(\d{1,2})[-\/](\d{1,2})[-\/](\d{4})$/.exec(s);
  if (m) return makeLocalDate_(Number(m[3]), Number(m[2]), Number(m[1]));

  // Last-resort parser for labels such as "10 Aug 2026".
  var d = new Date(s);
  if (!isNaN(d.getTime())) {
    return new Date(d.getFullYear(), d.getMonth(), d.getDate(), 0, 0, 0, 0);
  }
  return null;
}

function makeLocalDate_(year, month1, day) {
  if (!isFinite(year) || !isFinite(month1) || !isFinite(day)) return null;
  var d = new Date(year, month1 - 1, day, 0, 0, 0, 0);
  if (d.getFullYear() !== year || d.getMonth() !== month1 - 1 || d.getDate() !== day) return null;
  return d;
}

function rallyDayWindow_(dateLabel) {
  var d = parseRallyDateLabel_(dateLabel);
  if (!d) return null;
  var start = new Date(d.getFullYear(), d.getMonth(), d.getDate(), FORM_DAY_SWITCH_HOUR, 0, 0, 0);
  var end = new Date(d.getFullYear(), d.getMonth(), d.getDate() + 1, FORM_DAY_SWITCH_HOUR, 0, 0, 0);
  return { start: start, end: end };
}

function formatRallyDateForForm_(dateLabel) {
  var d = parseRallyDateLabel_(dateLabel);
  if (!d) return String(dateLabel || '').trim();
  var dd = d.getDate() < 10 ? '0' + d.getDate() : String(d.getDate());
  var mm = (d.getMonth() + 1) < 10 ? '0' + (d.getMonth() + 1) : String(d.getMonth() + 1);
  return dd + '-' + mm + '-' + d.getFullYear();
}

function applyFormDayWindow_(form, dayVal, dateLabel, nowValue) {
  if (!form) return;
  var window = rallyDayWindow_(dateLabel);
  if (!window) return; // Backward compatibility: no date = retain prior behaviour.

  var now = nowValue instanceof Date ? nowValue : new Date();
  var active = now.getTime() >= window.start.getTime() && now.getTime() < window.end.getTime();
  var label = formatRallyDateForForm_(dateLabel);

  if (active) {
    form.setAcceptingResponses(true);
    return;
  }

  var message;
  if (now.getTime() < window.start.getTime()) {
    message = 'Day ' + dayVal + ' (' + label + ') is not yet open. This form opens at 03:00 on the rally date.';
  } else {
    message = 'Day ' + dayVal + ' (' + label + ') is closed. Rally day changed at 03:00; please use the current day form.';
  }
  try { form.setCustomClosedFormMessage(message); } catch (_) {}
  form.setAcceptingResponses(false);
}

function ensureFormWindowBoundaryTrigger_(when) {
  if (!(when instanceof Date) || isNaN(when.getTime())) return;
  if (when.getTime() <= Date.now()) return;

  var props = getProps_();
  var key = FORM_WINDOW_TRIGGER_PREFIX + String(when.getTime());
  if (props.getProperty(key) === '1') return;

  ScriptApp.newTrigger('refreshRallyFormWindows_')
    .timeBased()
    .at(when)
    .create();
  props.setProperty(key, '1');
}

function scheduleFormDayWindow_(dateLabel) {
  var window = rallyDayWindow_(dateLabel);
  if (!window) return;
  ensureFormWindowBoundaryTrigger_(window.start);
  ensureFormWindowBoundaryTrigger_(window.end);
}

/**
 * One-off time trigger target. Re-evaluates every tagged STC/Marshal form, so one
 * 03:00 boundary trigger can close the old day and open the new day together.
 */
function refreshRallyFormWindows_() {
  var props = getProps_();
  var all = props.getProperties();
  var now = new Date();

  Object.keys(all).forEach(function (key) {
    if (key.indexOf(FORMTAG_PREFIX) !== 0) return;
    var tag;
    try { tag = JSON.parse(all[key]); } catch (_) { tag = null; }
    if (!tag || (String(tag.kind) !== 'STC' && String(tag.kind) !== 'MARSHAL')) return;
    if (!tag.dateLabel) return;

    var formId = key.substring(FORMTAG_PREFIX.length);
    try {
      var form = FormApp.openById(formId);
      applyFormDayWindow_(form, tag.day, tag.dateLabel, now);
    } catch (err) {
      try { logError_(tag.rallyName || '', 'refreshRallyFormWindows_', err, { formId: formId, day: tag.day, dateLabel: tag.dateLabel }); } catch (_) {}
    }
  });

  // Remove stale de-duplication markers after their boundary has passed.
  Object.keys(all).forEach(function (key) {
    if (key.indexOf(FORM_WINDOW_TRIGGER_PREFIX) !== 0) return;
    var ms = Number(key.substring(FORM_WINDOW_TRIGGER_PREFIX.length));
    if (isFinite(ms) && ms <= now.getTime()) {
      try { props.deleteProperty(key); } catch (_) {}
    }
  });
}

/**
 * Ensure a single base froot folder for all Rally Controls work.
 */
function ensureBaseFolder_() {
  var lock = LockService.getScriptLock();
  lock.waitLock(30000);
  try {

    // Dummy-proof root folder handling:
    // Prefer an explicit ROOT_FOLDER_ID stored in Script Properties.
    // Fall back to finding/creating a single folder named ROOT_FOLDER_NAME.
    var props = getProps_();

    var rootKey = PROP_PREFIX + 'rootFolderId';
    var folderId = props.getProperty(rootKey);
    var folder = null;

    if (folderId) {
      try {
        folder = rcDriveGetFolderById_(folderId);
      } catch (e) {
        folder = null;
      }
    }

    if (!folder) {
      // Try to find exactly one folder by name.
      var it = DriveApp.getFoldersByName(ROOT_FOLDER_NAME);
      var found = [];
      while (it.hasNext()) {
        found.push(it.next());
        if (found.length > 2) break; // enough to know it's not unique
      }

      if (found.length === 1) {
        folder = found[0];
      } else if (found.length === 0) {
        folder = rcRetry_('DriveApp.createFolder root', function () { return DriveApp.createFolder(ROOT_FOLDER_NAME); });
      } else {
        // More than one folder has the same name - require explicit ID.
        throw new Error('Multiple Drive folders named "' + ROOT_FOLDER_NAME + '" were found. ' +
          'Please keep only one, or set Script Property ' + rootKey + ' to the correct folder ID.');
      }

      // Persist for all future runs.
      props.setProperty(rootKey, folder.getId());
    }

    return folder;

  } finally {
    try { lock.releaseLock(); } catch (e) {}
  }
}

function autoParkMasterScriptOnce_() {
  var props = getProps_();
  var parkedKey = PROP_PREFIX + 'masterScriptParked';
  if (props.getProperty(parkedKey) === '1') return;

  var scriptId;
  try {
    scriptId = ScriptApp.getScriptId();
  } catch (e) {
    return; // Shouldn't happen, but don't block uploads.
  }

  var file;
  try {
    file = rcDriveGetFileById_(scriptId);
  } catch (e) {
    // Likely a bound script or no Drive scope yet.
    return;
  }

  // Ensure root folder exists (and IDs stored).
  var rootFolder = ensureBaseFolder_();

  // Rename if it's still the generic "Untitled project" or otherwise unhelpful.
  try {
    var currentName = file.getName() || '';
    if (/^Untitled project$/i.test(currentName) || /^Untitled$/i.test(currentName) || !currentName.trim()) {
      file.setName(MASTER_SCRIPT_NAME);
    }
  } catch (e) {
    // Ignore rename failures.
  }

  // Add to root folder if not already present.
  try {
    rcFolderAddFile_(rootFolder, file);
  } catch (e) {
    // Ignore add failures.
  }

  // Remove from My Drive root to reduce clutter (if present).
  try {
    var root = DriveApp.getRootFolder();
    var parents = file.getParents();
    while (parents.hasNext()) {
      var p = parents.next();
      if (p.getId && p.getId() === root.getId()) {
        root.removeFile(file);
        break;
      }
    }
  } catch (e) {
    // Ignore remove failures.
  }

  props.setProperty(parkedKey, '1');
}


/**
 * Slugify rally names so we can keep per-rally IDs in ScriptProperties.
 */
function slugRallyName_(name) {
  var raw = (name || 'Rally').toLowerCase();
  raw = raw.replace(/[^a-z0-9]+/g, '_');
  raw = raw.replace(/^_+|_+$/g, '');
  return raw || 'rally';
}


function normalizeRallyName_(name) {
  return String(name || '').trim();
}

function isValidRallyName_(name) {
  var s = normalizeRallyName_(name);
  if (!s) return false;
  if (s.length > 80) return false;

  // Reject values that look like internal function names or identifiers.
  if (s.indexOf('_') !== -1) return false;
  if (/^ensure[A-Za-z0-9]+_?$/.test(s)) return false;
  if (/Trigger/i.test(s)) return false;

  return true;
}

/**
 * Append a single error row for a given rally into that rally’s Errors sheet.
 */
function logError_(a, b, c, d) {
  // Supports BOTH calling styles:
  //   logError_(rallyName, where, err, context)
  //   logError_(where, err, context)   // legacy/internal convenience
  try {
    var rallyName = '';
    var where = '';
    var err = null;
    var context = null;

    // Detect signature
    if (typeof a === 'string' && (b && (b.message || b.stack) || typeof b === 'string') && c !== undefined && d === undefined) {
      // logError_(where, err, context)
      where = a;
      err = b;
      context = c;
    } else if (typeof a === 'string' && (b && (b.message || b.stack) || typeof b === 'string') && c === undefined) {
      // logError_(where, err)
      where = a;
      err = b;
      context = null;
    } else {
      // logError_(rallyName, where, err, context)
      rallyName = a || '';
      where = b || '';
      err = c;
      context = d || null;
    }

    // Best-effort rallyName recovery
    if (!rallyName && context && context.rallyName) rallyName = String(context.rallyName);
    if (!rallyName) {
      var last = PropertiesService.getScriptProperties().getProperty('LAST_RALLY_NAME') || '';
      if (last) rallyName = last;
    }

    // If we still don't have a rally name, do NOT create Drive assets just to log.
    if (!rallyName) {
      console.error('[ERROR]', where, err, context || '');
      return;
    }

    var storage = ensureRallyStorage_(rallyName);
    var sheet = storage.errorsSheet;

    var message = err && err.message ? String(err.message) : String(err);
    var stack = err && err.stack ? String(err.stack) : '';
    var ctx = context ? JSON.stringify(context) : '';

    sheet.appendRow([ new Date(), where || '', message, stack, ctx ]);
  } catch (logErr) {
    console.error('logError_ failed', logErr, 'original:', a, b, c, d);
  }
}

 * from the entries system.
 *
 * Sheet layouts:
 *   STC Day X:     Entrant | # | STC 1 M | STC 1 GPS | STC 2 M | STC 2 GPS | ...
 *   Marshal Day X: Entrant | # | Marshal 1 | Marshal 2 | ... | Marshal m
 */

function buildDaySheetsForControls_(ss, parsed, dayOnly) {
  if (!parsed || !Array.isArray(parsed.days)) {
    return;
  }
  parsed.days.forEach(function (d) {
    if (dayOnly && d && d.day != null && String(d.day) !== String(dayOnly)) { return; }
    if (!d) return;
    var dayVal = (d.day != null) ? String(d.day) : '';
    if (dayOnly && String(dayVal) !== String(dayOnly)) { return; }
    if (!dayVal) return;

    var controls = Array.isArray(d.controls) ? d.controls : [];
    var dayDateLabel = (d && (d.dateLabel || d.date || d.rallyDate)) ||
      (parsed && (parsed.dateLabel || parsed.rallyDate || parsed.date)) || '';
    var dayDateDisplay = formatRallyDateForForm_(dayDateLabel);

    var stcControls = controls.filter(function (c) {
      return c && c.type === 'stc' && c.enabled;
    });
    var marshalControls = controls.filter(function (c) {
      return c && (c.type === 'marshal' || c.type === 'open') && c.enabled;
    });

    // STC Day sheet
    if (stcControls.length > 0) {
      var stcSheetName = 'STC Day ' + dayVal;
      var stcSheet = ss.getSheetByName(stcSheetName);
      if (!stcSheet) {
        stcSheet = ss.insertSheet(stcSheetName);
      } else {
        stcSheet.clearContents();
      }
      var stcHeader = ['Entrant', '#'];
      for (var i = 0; i < stcControls.length; i++) {
        var lab = (stcControls[i] && stcControls[i].label != null) ? String(stcControls[i].label) : ('STC ' + (i + 1));
        // v3.00: each STC has two visible data columns so the sheet can be used
        // for direct testing/review before and after the PWA is live.
        stcHeader.push(lab + ' M');
        stcHeader.push(lab + ' GPS');
      }
      stcSheet.getRange(1, 1, 1, stcHeader.length).setValues([stcHeader]);

      // Pre-populate Entrant and # from uploaded entrants list (if present)
      if (parsed && Array.isArray(parsed.entrants) && parsed.entrants.length > 0) {
        var stcRows = parsed.entrants.map(function (e) {
          var name = (e && e.name != null) ? String(e.name) : '';
          var rallyNo = (e && e.rallyNo != null) ? String(e.rallyNo) : '';
          return [name, rallyNo];
        });
        stcSheet.getRange(2, 1, stcRows.length, 2).setValues(stcRows);
      }
    }

    // Marshal Day sheet
    if (marshalControls.length > 0) {
      var mSheetName = 'Marshal Day ' + dayVal;
      var mSheet = ss.getSheetByName(mSheetName);
      if (!mSheet) {
        mSheet = ss.insertSheet(mSheetName);
      } else {
        mSheet.clearContents();
      }
      var mHeader = ['#'];
      for (var j = 0; j < marshalControls.length; j++) {
        var c = marshalControls[j];
        var lab = (c && c.label != null) ? String(c.label) : (c && c.type === 'open' ? ('Open Control ' + (j + 1)) : ('Marshal ' + (j + 1)));
        mHeader.push(lab);
      }
      mSheet.getRange(1, 1, 1, mHeader.length).setValues([mHeader]);
    }
  });
}

}function ensureResponseTabNamed_(ss, desiredTabName) {
  if (!ss || !desiredTabName) return;
  try {
    // If already present, do nothing.
    if (ss.getSheetByName(desiredTabName)) return;

    // Prefer Google's response sheet naming patterns.
    var sheets = ss.getSheets();
    var candidate = null;

    // First, exact common names.
    var exactNames = ['Form Responses 1', 'Form responses 1', 'Form_Responses1', 'Form_Responses 1', 'Responses 1'];
    for (var i = 0; i < exactNames.length; i++) {
      var sh = ss.getSheetByName(exactNames[i]);
      if (sh) { candidate = sh; break; }
    }

    // Fallback: match "Form responses N" (choose the lowest N).
    if (!candidate) {
      var bestN = 9999;
      for (var j = 0; j < sheets.length; j++) {
        var name = sheets[j].getName();
        var mm = name.match(/^Form[\s_]*responses[\s_]*(\d+)$/i);
        if (mm) {
          var n = parseInt(mm[1], 10);
          if (!isNaN(n) && n < bestN) {
            bestN = n;
            candidate = sheets[j];
          }
        }
      }
    }

    if (candidate) {
      candidate.setName(desiredTabName);
    }
  } catch (_) {}
}



function pruneOldFormResponseTabs_(ss, desiredTabName) {
  // Keep the newest "Form responses N" sheet and remove older ones,
  // also removing any previous desiredTabName sheet so we end with ONE clean tab.
  if (!ss) return;
  try {
    var sheets = ss.getSheets();
    var newest = null;
    var newestN = -1;

    // Identify newest response sheet by highest suffix N.
    for (var i = 0; i < sheets.length; i++) {
      var nm = sheets[i].getName();
      var mm = nm.match(/^Form[\s_]*responses[\s_]*(\d+)$/i);
      if (mm) {
        var n = parseInt(mm[1], 10);
        if (!isNaN(n) && n > newestN) { newestN = n; newest = sheets[i]; }
      }
    }

    // If none matched, nothing to prune.
    if (!newest) return;

    // Delete any old renamed response tab (desiredTabName) if it's not the newest sheet.
    if (desiredTabName) {
      var oldDesired = ss.getSheetByName(desiredTabName);
      if (oldDesired && oldDesired.getSheetId() !== newest.getSheetId()) {
        ss.deleteSheet(oldDesired);
      }
    }

    // Delete all other "Form responses N" sheets except newest.
    sheets = ss.getSheets(); // refresh after possible delete
    for (var j = 0; j < sheets.length; j++) {
      var name2 = sheets[j].getName();
      var m2 = name2.match(/^Form[\s_]*responses[\s_]*(\d+)$/i);
      if (m2 && sheets[j].getSheetId() !== newest.getSheetId()) {
        ss.deleteSheet(sheets[j]);
      }
    }

    // Rename newest to desiredTabName if provided.
    if (desiredTabName) {
      try { newest.setName(desiredTabName); } catch (_) {}
    }
  } catch (_) {}
}


// --- Deferred response-sheet cleanup (avoids races after relinking) ---

function queueResponseSheetCleanup_(spreadsheetId, desiredTabName) {
  if (!spreadsheetId) return;
  try {
    var props = PropertiesService.getScriptProperties();
    var key = 'RC_RESP_CLEANUP_QUEUE';
    var raw = props.getProperty(key);
    var arr = [];
    if (raw) {
      try { arr = JSON.parse(raw); } catch (_) { arr = []; }
    }

    // De-dupe by spreadsheetId+desiredTabName.
    var found = false;
    for (var i = 0; i < arr.length; i++) {
      if (arr[i] && arr[i].ssid === spreadsheetId && arr[i].tab === desiredTabName) { found = true; break; }
    }
    if (!found) arr.push({ ssid: spreadsheetId, tab: desiredTabName, t: Date.now() });

    props.setProperty(key, JSON.stringify(arr));

    // Ensure we have exactly one near-term trigger.
    var triggers = ScriptApp.getProjectTriggers();
    var have = false;
    for (var j = 0; j < triggers.length; j++) {
      if (triggers[j].getHandlerFunction && triggers[j].getHandlerFunction() === 'runQueuedResponseSheetCleanups_') {
        have = true; break;
      }
    }
    if (!have) {
      // Run shortly after upload so the new "Form responses N" tab has time to appear.
      ScriptApp.newTrigger('runQueuedResponseSheetCleanups_').timeBased().after(60 * 1000).create();
    }
  } catch (_) {}
}

function runQueuedResponseSheetCleanups_() {
  var key = 'RC_RESP_CLEANUP_QUEUE';
  try {
    var props = PropertiesService.getScriptProperties();
    var raw = props.getProperty(key);
    if (!raw) {
      cleanupSelfTrigger_('runQueuedResponseSheetCleanups_');
      return;
    }
    var arr = [];
    try { arr = JSON.parse(raw); } catch (_) { arr = []; }

    for (var i = 0; i < arr.length; i++) {
      var job = arr[i];
      if (!job || !job.ssid) continue;
      try {
        var ss = SpreadsheetApp.openById(job.ssid);
        var desired = job.tab || null;
        pruneOldFormResponseTabs_(ss, desired);
        if (desired) ensureResponseTabNamed_(ss, desired);
      } catch (_) {}
    }

    // Clear queue and remove trigger.
    props.deleteProperty(key);
    cleanupSelfTrigger_('runQueuedResponseSheetCleanups_');
  } catch (_) {
    // Don't loop forever; clear queue on hard error.
    try { PropertiesService.getScriptProperties().deleteProperty(key); } catch (_) {}
    cleanupSelfTrigger_('runQueuedResponseSheetCleanups_');
  }
}

function cleanupSelfTrigger_(handlerName) {
  try {
    var triggers = ScriptApp.getProjectTriggers();
    for (var i = 0; i < triggers.length; i++) {
      if (triggers[i].getHandlerFunction && triggers[i].getHandlerFunction() === handlerName) {
        ScriptApp.deleteTrigger(triggers[i]);
      }
    }
  } catch (_) {}
}


function responseTabNameFromFileName_(responseSheetFileName) {
  var s = (responseSheetFileName || '').toString();

  // Entries
  if (/^Entries\s+Responses\b/i.test(s)) return 'Entries Responses';

  // STC / Marshal with Day number if present
  var dayMatch = s.match(/\bDay\s*(\d+)\b/i);
  var dayPart = dayMatch ? (' D' + dayMatch[1]) : '';
  if (/\bSTC\b/i.test(s)) return 'STC' + dayPart + ' Responses';
  if (/\bMarshal\b/i.test(s)) return 'Marshal' + dayPart + ' Responses';

  // Generic fallback
  if (s) return s;
  return 'Responses';
}



function safeDeleteAllResponses_(form) {
  if (!form) return;
  try {
    if (typeof form.deleteAllResponses === 'function') {
      form.deleteAllResponses();
    }
  } catch (e) {
    // Non-fatal: we can continue even if old responses couldn't be cleared.
  }
}

function ensureFormHasResponseSheet_(rallyFolder, form, responseSheetName) {
  if (!form) return;

  // Each time the form schema is rebuilt, Google can append new columns to the existing
  // response sheet because item IDs change. The cleanest "slate refresh" is:
  // - relink destination (forces a fresh "Form responses N" tab)
  // - delete older "Form responses" tabs and keep only the newest (no trash, no clutter)
  var destId = null;
  try {
    destId = (typeof form.getDestinationId === 'function') ? form.getDestinationId() : null;
  } catch (e) {
    destId = null;
  }

  // Ensure we have a destination spreadsheet in the rally folder.
  var respSs = null;
  if (destId) {
    try { respSs = SpreadsheetApp.openById(destId); } catch (_) { respSs = null; }
  }
  if (!respSs) {
    respSs = findOrCreateSpreadsheetInFolder_(rallyFolder, responseSheetName);
  }

  // Relink destination to force a fresh response tab.
  try {
    if (typeof form.removeDestination === 'function') form.removeDestination();
  } catch (_) {}

  form.setDestination(FormApp.DestinationType.SPREADSHEET, respSs.getId());

  // Queue a post-link cleanup. Google sometimes creates/attaches the new response tab asynchronously,
// so pruning immediately can miss the freshly-created tab. We queue a short delayed cleanup
// (no Drive trash; leaves ONE response tab).
  var desiredTab = responseTabNameFromFileName_(responseSheetName);
  queueResponseSheetCleanup_(respSs.getId(), desiredTab);

  // Best-effort immediate prune/rename (may be finalized by delayed cleanup).
  try { pruneOldFormResponseTabs_(respSs, desiredTab); } catch (_) {}
  try { ensureResponseTabNamed_(respSs, desiredTab); } catch (_) {}
}

function getMaxCarsFromParsed_(parsed) {
  if (!parsed) return 30;
  // Prefer explicit entrantsCount if provided by the upload JSON.
  if (typeof parsed.entrantsCount === 'number' && parsed.entrantsCount > 0) {
    return parsed.entrantsCount;
  }
  // Fallback: if an entries array is provided, use its length.
  if (Array.isArray(parsed.entries) && parsed.entries.length > 0) {
    return parsed.entries.length;
  }
  // Final fallback for older uploads / R&D: keep previous default.
  return 30;
}

 * buildFormsForControls_:
 * - For each day in parsed.days, builds/updates per-day STC and Marshal forms.
 * - Forms are created inside the rally folder and pointed at per-form response spreadsheets (NOT the rally spreadsheet).
 * - Question sets follow the spec agreed for v1.09.
 */
function buildFormsForControls_(rallyFolder, ss, rallyName, parsed, dayOnly) {
  if (!rallyFolder || !ss) return;
  if (!parsed || !Array.isArray(parsed.days)) return;

  var rallyTitle = rallyName || ss.getName();

  parsed.days.forEach(function (d) {
    if (!d) return;
    var dayVal = (d.day != null) ? String(d.day) : '';
    if (!dayVal) return;
    if (dayOnly && String(dayVal) !== String(dayOnly)) { return; }

    var controls = Array.isArray(d.controls) ? d.controls : [];
    var dayDateLabel = (d && (d.dateLabel || d.date || d.rallyDate)) ||
      (parsed && (parsed.dateLabel || parsed.rallyDate || parsed.date)) || '';
    var dayDateDisplay = formatRallyDateForForm_(dayDateLabel);

    var stcControls = controls.filter(function (c) {
      return c && c.type === 'stc' && c.enabled;
    });
    var marshalControls = controls.filter(function (c) {
      return c && (c.type === 'marshal' || c.type === 'open') && c.enabled;
    });

    if (stcControls.length === 0 && marshalControls.length === 0) {
      return;
    }

    // --- STC form per day ---
    if (stcControls.length > 0) {
      var stcDriveName = 'STC – Day ' + dayVal + ' – ' + rallyTitle;
      var stcFormTitle = rallyTitle + ' – STC Times – Day ' + dayVal + (dayDateDisplay ? ' – ' + dayDateDisplay : '');
      var stcForm = findOrCreateFormInFolder_(rallyFolder, stcDriveName);
      stcForm.setTitle(stcFormTitle);
      stcForm.setDescription('');
      tagForm_(stcForm, 'STC', dayVal, rallyTitle, rallyFolder.getId(), dayDateLabel);
      safeDeleteAllResponses_(stcForm);

      // Clear existing items
      var stcItems = stcForm.getItems();
      for (var i = stcItems.length - 1; i >= 0; i--) {
        stcForm.deleteItem(stcItems[i]);
      }

      // 1) Rally Number (required, numeric)
      var rallyNumItem = stcForm.addTextItem();
      rallyNumItem.setTitle('Rally Number').setRequired(true);
      try {
        var numValidation = FormApp.createTextValidation()
          .requireTextMatchesPattern('^(\\d+|[Oo][Cc]\\d+)$')
          .build();
        rallyNumItem.setValidation(numValidation);
      } catch (eNum) {
        console.warn('Could not set number validation for Rally Number', eNum);
      }

      // 2) Driver Surname (required)
      var surnameItem = stcForm.addTextItem();
      surnameItem.setTitle('Driver Surname').setRequired(true);

      // 3) How to enter times (section header with help text)
      var infoItem = stcForm.addSectionHeaderItem();
      infoItem.setTitle('Time Entries');
      infoItem.setHelpText('Enter digits only. Do not use :, . or ,.\nLeading zeros are not required.\nExamples:\n15 = 00:00:15\n128 = 00:01:28\n93545 = 09:35:45\n143012 = 14:30:12');

      // 4) STC 1..n time fields (required)
      stcControls.forEach(function (c, idx) {
        var stcItem = stcForm.addTextItem();
        stcItem.setTitle((c && c.label) ? String(c.label) : ('STC ' + (idx + 1)));
        stcItem.setRequired(true);
        applyDigitsOnlyTimeValidation_(stcItem);

      stcCheckItem.setTitle('I have checked my times');
      stcCheckItem.setChoiceValues(['Yes']);
      stcCheckItem.setRequired(true);

      // Destination: rally spreadsheet
      // Only set destination the first time; repeated calls can create extra
      // "Form responses" tabs. If the form already has a destination, reuse it.
      ensureFormHasResponseSheet_(rallyFolder, stcForm, 'STC Responses – Day ' + dayVal + ' – ' + rallyTitle);

      ensureFormSubmitTriggerForForm_(stcForm, 'handleStcSubmit_AllDays_');
      applyFormDayWindow_(stcForm, dayVal, dayDateLabel);
      scheduleFormDayWindow_(dayDateLabel);
    }

    // --- Marshal form per day ---
    if (marshalControls.length > 0) {
      var marshalDriveName = 'Marshals – Day ' + dayVal + ' – ' + rallyTitle;
      var marshalFormTitle = rallyTitle + ' – Marshal – Day ' + dayVal + (dayDateDisplay ? ' – ' + dayDateDisplay : '');
      var marshalForm = findOrCreateFormInFolder_(rallyFolder, marshalDriveName);
      marshalForm.setTitle(marshalFormTitle);
      marshalForm.setDescription('');
      tagForm_(marshalForm, 'MARSHAL', dayVal, rallyTitle, rallyFolder.getId(), dayDateLabel);
      safeDeleteAllResponses_(marshalForm);

      // Clear existing items
      var mItems = marshalForm.getItems();
      for (var j = mItems.length - 1; j >= 0; j--) {
        marshalForm.deleteItem(mItems[j]);
      }

      // 1) Marshal Number (required, numeric)
      var marshalNumItem = marshalForm.addTextItem();
      marshalNumItem.setTitle('Control Number').setRequired(true);
      marshalNumItem.setHelpText('Enter a Marshal number (e.g. 1, 2) or an Open Control number (e.g. OC1).');
      try {
        var mNumValidation = FormApp.createTextValidation()
          .requireTextMatchesPattern('^(\\d+|[Oo][Cc]\\d+)$')
          .build();
        marshalNumItem.setValidation(mNumValidation);
      } catch (eMN) {
        Logger.log('WARN: marshal number validation not applied: ' + eMN);
      }

            // 2) Marshal/OC Times (read-only info block; populated from upload JSON if present)
      // - Uses parsed.marshalWindowsText (preferred) and optional parsed.openWindowsText
      // - Or builds from parsed.marshalWindows[] and optional parsed.openWindows[]
      try {
        // Build ordered list for ETA windows in true control sequence (marshal/open interleaved)
        var lines = [];

        // Lookup tables from parsed windows (if provided)
        var mWinByNum = {};
        var ocWinByNum = {};

        if (parsed && Array.isArray(parsed.marshalWindows)) {
          parsed.marshalWindows.forEach(function (w) {
            if (!w) return;
            var wl = (w.label != null ? String(w.label) : '').trim();
            var wid = (w.id != null ? String(w.id) : '').trim();
            var numMatch = (wl.match(/(\d+)/) || wid.match(/(\d+)/));
            var num = numMatch ? String(numMatch[1]) : '';

              firstEta: (w.firstEta != null ? String(w.firstEta) : '').trim(),
              lastEta:  (w.lastEta  != null ? String(w.lastEta)  : '').trim()
            };
          });
        }

        if (parsed && Array.isArray(parsed.openWindows)) {
          parsed.openWindows.forEach(function (w) {
            if (!w) return;
            var wl = (w.label != null ? String(w.label) : '').trim();
            var wid = (w.id != null ? String(w.id) : '').trim(); // e.g. oc1
            var numMatch = (wl.match(/(\d+)/) || wid.match(/(\d+)/));
            var num = numMatch ? String(numMatch[1]) : '';
            if (!num) return;
            ocWinByNum[num] = {
              firstEta: (w.firstEta != null ? String(w.firstEta) : '').trim(),
              lastEta:  (w.lastEta  != null ? String(w.lastEta)  : '').trim()
            };
          });
        }

        var seq = marshalControls.filter(function (c) {
          return c && c.enabled && (c.type === 'marshal' || c.type === 'open');
        });

        seq.forEach(function (c, idx) {
          var clabel = (c && c.label != null ? String(c.label) : '').trim();
          var cid = (c && c.id != null ? String(c.id) : '').trim();

          // Determine number from label/id; fallback to sequential index
          var numMatch = (clabel.match(/(\d+)/) || cid.match(/(\d+)/));
          var num = numMatch ? String(numMatch[1]) : String(idx + 1);

          var label = clabel;
          if (!label) {
            if (c.type === 'marshal') label = 'Marshal ' + num;
            if (c.type === 'open')   label = 'Open Control ' + num;
          }

          if (c.type === 'marshal') {
            var mw = mWinByNum[num];
            if (mw && (mw.firstEta || mw.lastEta)) {
              lines.push(label + ' — ' + (mw.firstEta || '') + ' – ' + (mw.lastEta || ''));
            } else {
              lines.push(label + ' — no ETA (after START STC)');
            }
            return;
          }

          if (c.type === 'open') {
            var ow = ocWinByNum[num];
            if (ow && (ow.firstEta || ow.lastEta)) {
              lines.push(label + ' — ' + (ow.firstEta || '') + ' – ' + (ow.lastEta || ''));
            } else {
              lines.push(label + ' — no ETA available');
            }
            return;
          }
        });

        var combinedTimesText = lines.join('\n');
        if (combinedTimesText) {
          var mTimesItem = marshalForm.addSectionHeaderItem();
          mTimesItem.setTitle('ETA time of cars at Marshal/Open Control location');
          mTimesItem.setHelpText(combinedTimesText);
        }
      } catch (eMT) {
        console.warn('Could not add Marshal Times block', eMT);
      }

// 3) How to enter times (section header with help text)
      var mInfoItem = marshalForm.addSectionHeaderItem();
      mInfoItem.setTitle('Time Entries');
      mInfoItem.setHelpText('Enter digits only. Do not use :, . or ,.\nLeading zeros are not required.\nExamples:\n15 = 00:00:15\n128 = 00:01:28\n93545 = 09:35:45\n143012 = 14:30:12');

      // 3) Car 1..N Time fields (required) - based on entrants count from upload (fallback 30)
      var maxCars = getMaxCarsFromParsed_(parsed);
      for (var carNo = 1; carNo <= maxCars; carNo++) {
        var carItem = marshalForm.addTextItem();
        carItem.setTitle('Car ' + carNo + ' Time');
        carItem.setRequired(true);
        applyDigitsOnlyTimeValidation_(carItem);
      }

      // 4) I have checked the times (checkbox, required)
      var mCheckItem = marshalForm.addCheckboxItem();
      mCheckItem.setTitle('I have checked the times');
      mCheckItem.setChoiceValues(['Yes']);
      mCheckItem.setRequired(true);

      // Destination: rally spreadsheet
      // Only set destination the first time; repeated calls can create extra
      // "Form responses" tabs. If the form already has a destination, reuse it.
      ensureFormHasResponseSheet_(rallyFolder, marshalForm, 'Marshal Responses – Day ' + dayVal + ' – ' + rallyTitle);

      ensureFormSubmitTriggerForForm_(marshalForm, 'handleMarshalSubmit_AllDays_');
      applyFormDayWindow_(marshalForm, dayVal, dayDateLabel);
      scheduleFormDayWindow_(dayDateLabel);

    }
  });
}

 * for a given rally/day. Note: entrantFormUrl in the return value is used
 * by the UI for historical reasons and currently carries the STC form URL.
 */
function initForms_(options) {
  options = options || {};
  var rallyName = options.rallyName || 'Rally';
  var day = options.day || '1';
  var dayVal = String(day || '1');

  try {
    var storage = ensureRallyStorage_(rallyName);
    var rallyFolder = storage.rallyFolder;
    var ss = storage.spreadsheet;
    var rallyTitle = storage.rallyName;

    // Prefer the per-day STC / Marshal forms that were built from the
    // uploaded controls configuration (buildFormsForControls_).
    var stcDriveName = 'STC – Day ' + dayVal + ' – ' + rallyTitle;
    var marshalDriveName = 'Marshals – Day ' + dayVal + ' – ' + rallyTitle;

    var stcForm = null;
    var marshalForm = null;

    var stcFiles = rallyFolder.getFilesByName(stcDriveName);
    if (stcFiles.hasNext()) {
      var stcFile = stcFiles.next();
      stcForm = FormApp.openById(stcFile.getId());
    }

    var marshalFiles = rallyFolder.getFilesByName(marshalDriveName);
    if (marshalFiles.hasNext()) {
      var marshalFile = marshalFiles.next();
      marshalForm = FormApp.openById(marshalFile.getId());
    }

    if (!stcForm && !marshalForm) {
      throw new Error('No STC/Marshal forms found for Day ' + dayVal +
        ' in rally "' + rallyTitle + '". Upload controls first.');
    }

    return {
      rallyName: storage.rallyName,
      entrantFormUrl: stcForm ? stcForm.getPublishedUrl() : '',
      marshalFormUrl: marshalForm ? marshalForm.getPublishedUrl() : '',
      spreadsheetUrl: ss.getUrl()
    };
  } catch (err) {
    logError_(rallyName, 'initForms_', err, options);
    throw err;
  }
}

function normalizeTimingMode_(mode) {
  var s = String(mode || '').trim().toLowerCase();
  if (s === 'gps') return 'GPS';
  return 'Manual';
}

function serializeGpsLine_(gpsLine) {
  if (gpsLine === undefined || gpsLine === null || gpsLine === '') return '';
  try {
    if (typeof gpsLine === 'string') {
      var trimmed = gpsLine.trim();
      if (!trimmed) return '';
      // Store valid JSON compactly when possible; otherwise preserve the string.
      try { return JSON.stringify(JSON.parse(trimmed)); } catch (_) { return trimmed; }
    }
    return JSON.stringify(gpsLine);
  } catch (_) {
    return '';
  }
}

function parseGpsLine_(raw) {
  if (raw === undefined || raw === null || raw === '') return null;
  try {
    if (typeof raw === 'object') return raw;
    var s = String(raw || '').trim();
    if (!s) return null;
    return JSON.parse(s);
  } catch (_) {
    return null;
  }
}

function rcTrackingDayNumber_(value) {
  var m = String(value || '').match(/\d+/);
  return m ? String(Number(m[0])) : String(value || '').trim().toLowerCase();
}

function rcTrackingRouteLabel_(value) {
  var s = String(value || '').trim();
  if (!s) return 'Route 1';
  var m = s.match(/\d+/);
  return m ? ('Route ' + Number(m[0])) : s;
}

function rcTrackingLatestEventFor_(rallyTitle, day, route) {
  try {
    var storage = rcTrackingFindEventStorage_(rallyTitle);
    if (!storage || !storage.eventsSheet) return null;
    var values = storage.eventsSheet.getDataRange().getValues();
    var wantedTitle = String(rallyTitle || '').trim().toLowerCase();
    var wantedDay = rcTrackingDayNumber_(day);
    var wantedRoute = rcTrackingRouteLabel_(route).toLowerCase();
    for (var i = values.length - 1; i >= 1; i--) {
      var rowTitle = String(values[i][3] || '').trim().toLowerCase();
      var rowDay = rcTrackingDayNumber_(values[i][4]);
      var rowRoute = rcTrackingRouteLabel_(values[i][5]).toLowerCase();
      if (wantedTitle && rowTitle !== wantedTitle) continue;
      if (wantedDay && rowDay && rowDay !== wantedDay) continue;
      if (wantedRoute && rowRoute && rowRoute !== wantedRoute) continue;
      return {
        eventId: String(values[i][0] || '').trim(),
        rallyTitle: String(values[i][3] || '').trim(),
        day: String(values[i][4] || '').trim(),
        route: rcTrackingRouteLabel_(values[i][5]),
        routeData: rallyTrackingParseJson_(values[i][8], {})
      };
    }
  } catch (_) {}
  return null;
}

function rcTrackingFindWaypointArray_(node, depth) {
  if (!node || depth > 8) return null;
  if (Array.isArray(node)) {
    if (node.length && node.some(function (p) {
      return p && typeof p === 'object' && Number.isFinite(Number(p.lat !== undefined ? p.lat : p.latitude)) && Number.isFinite(Number(p.lng !== undefined ? p.lng : p.longitude));
    })) return node;
    for (var i = 0; i < node.length; i++) {
      var nestedArray = rcTrackingFindWaypointArray_(node[i], depth + 1);
      if (nestedArray) return nestedArray;
    }
    return null;
  }
  if (typeof node !== 'object') return null;
  var preferred = ['waypoints', 'routePoints', 'points'];
  for (var p = 0; p < preferred.length; p++) {
    if (Array.isArray(node[preferred[p]]) && node[preferred[p]].length) return node[preferred[p]];
  }
  var keys = Object.keys(node);
  for (var k = 0; k < keys.length; k++) {
    var nested = rcTrackingFindWaypointArray_(node[keys[k]], depth + 1);
    if (nested) return nested;
  }
  return null;
}

function rcTrackingEndOfEvent_(routeData) {
  var points = rcTrackingFindWaypointArray_(routeData, 0) || [];
  for (var i = points.length - 1; i >= 0; i--) {
    var point = points[i] || {};
    var lat = Number(point.lat !== undefined ? point.lat : point.latitude);
    var lon = Number(point.lng !== undefined ? point.lng : point.longitude);
    if (!Number.isFinite(lat) || !Number.isFinite(lon)) continue;
    return {
      enabled: true,
      latitude: lat,
      longitude: lon,
      detectionGateM: 200,
      radiusM: 30
    };
  }
  return { enabled: false, latitude: null, longitude: null, detectionGateM: 200, radiusM: 30 };
}

  };

/**
 * Return a read-only STC Recorder PWA configuration for one rally/day.
 * This uses existing ControlsConfig data only; it does not create folders,
 * sheets, forms, or modify existing data.
 */
function stcPwaConfig_(options) {
  var rallyName = (options && options.rallyName) || 'Rally';
  var day = String((options && options.day) || '1');

  var ss = findRallySpreadsheet_(rallyName);
  if (!ss) {
    return {
      ok: false,
      mode: 'stcPwaConfig',
      error: 'No rally spreadsheet found. Upload controls first.',
      rallyTitle: rallyName,
      dayLabel: 'Day ' + day,
      date: '',
      pwaUiMode: normalizePwaUiMode_(STC_PWA_UI_MODE_FALLBACK),
      controls: []
    };
  }

  var sheet = ss.getSheetByName(CONTROLS_SHEET_NAME);
  if (!sheet) {
    return {
      ok: false,
      mode: 'stcPwaConfig',
      error: 'ControlsConfig sheet not found. Upload controls first.',
      rallyTitle: rallyName,
      dayLabel: 'Day ' + day,
      date: '',
      pwaUiMode: normalizePwaUiMode_(STC_PWA_UI_MODE_FALLBACK),
      controls: []
    };
  }

  var lastRow = sheet.getLastRow();
  var lastCol = sheet.getLastColumn();
  if (lastRow < 2 || lastCol < 1) {
    return {
      ok: false,
      mode: 'stcPwaConfig',
      error: 'No controls data found. Upload controls first.',
      rallyTitle: rallyName,
      dayLabel: 'Day ' + day,
      date: '',
      pwaUiMode: normalizePwaUiMode_(STC_PWA_UI_MODE_FALLBACK),
      controls: []
    };
  }

  var values = sheet.getRange(1, 1, lastRow, lastCol).getDisplayValues();
  var header = values[0] || [];

  function col_(name, fallback) {
    for (var i = 0; i < header.length; i++) {
      if (String(header[i]).trim().toLowerCase() === String(name).trim().toLowerCase()) return i;
    }
    return fallback;
  }

  var dayCol = col_('Day', 0);
  var idCol = col_('ControlId', 1);
  var typeCol = col_('Type', 2);
  var labelCol = col_('Label', 3);
  var enabledCol = col_('Enabled', 4);
  var dateCol = col_('DateLabel', col_('Date', col_('DayDate', -1)));
  var timingModeCol = col_('TimingMode', -1);
  var gpsLineCol = col_('GpsLine', col_('GPSLine', -1));

  var controls = [];
  var dateLabel = '';
  var timingMode = 'Manual';
  var pwaUiMode = normalizePwaUiMode_(STC_PWA_UI_MODE_FALLBACK);

  for (var r = 1; r < values.length; r++) {
    var row = values[r];
    if (String(row[dayCol] || '') !== day) continue;

    var rawType = String(row[typeCol] || '').trim().toLowerCase();
    var label = String(row[labelCol] || '').trim();
    var enabledText = String(row[enabledCol] || '').trim().toLowerCase();
    var enabled = (enabledText === 'true' || enabledText === 'yes' || enabledText === '1' || enabledText === 'enabled');

    if (!enabled) continue;
    if (rawType !== 'stc') continue;
    if (!label) label = 'STC ' + (controls.length + 1);

    if (!dateLabel && dateCol >= 0 && row[dateCol]) dateLabel = String(row[dateCol]).trim();
    if (timingModeCol >= 0 && row[timingModeCol]) {
      timingMode = normalizeTimingMode_(row[timingModeCol]);
      pwaUiMode = pwaUiModeFromTimingMode_(timingMode);
    }

    var isStart = /^start\s+stc\b/i.test(label);
    var gpsLine = (gpsLineCol >= 0) ? parseGpsLine_(row[gpsLineCol]) : null;
    var controlObj = {
      id: String(row[idCol] || (controls.length + 1)),
      label: label,
      type: isStart ? 'startStc' : 'stc'
    };
    if (gpsLine) controlObj.gpsLine = gpsLine;
    controls.push(controlObj);
  }

  var submitUrl = '';
  try { submitUrl = ScriptApp.getService().getUrl() || ''; } catch (_) { submitUrl = ''; }

  var config = {
    configVersion: 1,
    rallyTitle: rallyName,
    dayLabel: 'Day ' + day,
    date: dateLabel,
    timingMode: timingMode,
    pwaUiMode: pwaUiMode,
    submitUrl: submitUrl,
    trackingEnabled: false,
    tracking: rcTrackingPwaConfig_(rallyName, day, submitUrl),
    controls: controls
  };

  var pwaBaseUrl = String(STC_PWA_BASE_URL || '').trim();
  var pwaUrl = '';
  if (pwaBaseUrl) {
    var sep = (pwaBaseUrl.indexOf('?') === -1) ? '?' : '&';
    pwaUrl = pwaBaseUrl + sep + 'config=' + encodeURIComponent(JSON.stringify(config));
  }

  return {
    ok: true,
    mode: 'stcPwaConfig',
    configVersion: config.configVersion,
    rallyTitle: config.rallyTitle,
    dayLabel: config.dayLabel,
    date: config.date,
    timingMode: config.timingMode,
    pwaUiMode: config.pwaUiMode,
    submitUrl: submitUrl,
    trackingEnabled: config.trackingEnabled,
    tracking: config.tracking,
    controls: config.controls,
    pwaBaseUrl: pwaBaseUrl,
    pwaUrl: pwaUrl
  };
}

    day: day,

  function readSheet_(name) {
    var sh = ss.getSheetByName(name);
    if (!sh) {
      return null;
    }
    var lastRow = sh.getLastRow();
    var lastCol = sh.getLastColumn();
    if (lastRow < 1 || lastCol < 1) {
      return null;
    }
    var values = sh.getRange(1, 1, lastRow, lastCol).getDisplayValues();
    if (!values || values.length === 0) {
      return null;
    }
    var header = values[0];
    var rows = [];
    if (values.length > 1) {
      for (var r = 1; r < values.length; r++) {
        rows.push(values[r]);
      }
    }
    return {
      sheetName: name,
      header: header,
      rows: rows
    };
  }

  result.stc = readSheet_(stcSheetName);
  result.marshal = readSheet_(marshalSheetName);

  return result;
}

/**
 * Handle Entry form submissions and keep a single row per entrant
 * based on Driver surname + Driver email.
 */
function handleEntrySubmit_(e) {
  var rallyName = '';
  var formId = '';
  var formTitle = '';
  try {
    var form = e && e.source;
    if (!form || typeof form.getTitle !== 'function') {
      console.error('[handleEntrySubmit_] Missing or invalid Form event source');
      return;
    }

    try { formId = form.getId ? String(form.getId()) : ''; } catch (_) { formId = ''; }
    try { formTitle = (form.getTitle() || '').toString().trim(); } catch (_) { formTitle = ''; }

    // Preferred routing: stable metadata stored by Form ID.
    var formTag = formId ? getFormTag_(formId) : null;
    if (formTag && String(formTag.kind || '') === 'entries') {
      rallyName = String(formTag.rallyName || '').trim();
    }

    // Legacy fallback: watermark in form description.
    if (!rallyName) {
      try {
        var desc = (form.getDescription && form.getDescription()) ? String(form.getDescription()) : '';
        var m = desc.match(/RC_RALLY\s*=\s*([^\n\r]+)/);
        if (m && m[1]) rallyName = String(m[1]).trim();
      } catch (_) {}
    }

    // Final fallback for forms created before v3.05.
    if (!rallyName) rallyName = formTitle;
    if (!rallyName) {
      console.error('[handleEntrySubmit_] Rally identity could not be resolved', JSON.stringify({ formId: formId, formTitle: formTitle }));
      return;
    }

    // Always write into the working Rally Controls spreadsheet for this rally.
    var ssAdmin = findRallySpreadsheet_(rallyName);
    if (!ssAdmin) {
      logError_(rallyName, 'handleEntrySubmit_', new Error('Rally spreadsheet not found'), {
        rallyName: rallyName,
        formId: formId,
        formTitle: formTitle
      });
      return;
    }

    var entriesSheetName = 'Admin - Entries';
    var sheet = ssAdmin.getSheetByName(entriesSheetName);

    if (!sheet) sheet = ssAdmin.insertSheet(entriesSheetName);
    if (sheet.getLastRow() < 1) {
      sheet.getRange(1, 1, 1, 9).setValues([[
        'Driver first name',
        'Driver surname',
        'Navigator first name',
        'Navigator surname',
        'Driver mobile number',
        'Driver email',
        'Car make/model',
        'Car year',
        'SG'
      ]]);
    }
    try {
      sheet.setColumnWidths(1, 7, 180);
      sheet.setColumnWidth(8, 100);
      sheet.setColumnWidth(9, 60);
      sheet.getRange('E:E').setNumberFormat('@');
    } catch (_) {}

    var itemResponses = (e && e.response && e.response.getItemResponses)
      ? e.response.getItemResponses()
      : [];
    if (!itemResponses.length) {
      logError_(rallyName, 'handleEntrySubmit_', new Error('Form submission contained no item responses'), {
        rallyName: rallyName,
        formId: formId,
        formTitle: formTitle
      });
      return;
    }

    var byTitle = {};
    for (var i = 0; i < itemResponses.length; i++) {
      var ir = itemResponses[i];
      var title = ir.getItem().getTitle();
      byTitle[title] = ir.getResponse();
    }

    var driverSurname = (byTitle['Driver surname'] || '').toString().trim();
    var driverEmail = (byTitle['Driver email'] || '').toString().trim();
    if (!driverSurname || !driverEmail) {
      logError_(rallyName, 'handleEntrySubmit_', new Error('Required entrant key is missing'), {
        rallyName: rallyName,
        formId: formId,
        formTitle: formTitle,
        hasDriverSurname: !!driverSurname,
        hasDriverEmail: !!driverEmail
      });
      return;
    }

    var rowValues = [
      (byTitle['Driver first name'] || '').toString().trim(),
      driverSurname,
      (byTitle['Navigator first name'] || '').toString().trim(),
      (byTitle['Navigator surname'] || '').toString().trim(),
      (byTitle['Driver mobile number'] || '').toString().trim(),
      driverEmail,
      (byTitle['Car make/model'] || '').toString().trim(),
      (byTitle['Car year'] || '').toString().trim(),
      (byTitle['Speed group'] || '').toString().trim()
    ];

    // Find existing row by Driver surname + Driver email; latest submission wins.
    var lastRow = sheet.getLastRow();
    var targetRow = lastRow + 1;
    var key = (driverSurname + '||' + driverEmail).toLowerCase();

    if (lastRow >= 2) {
      var dataRange = sheet.getRange(2, 1, lastRow - 1, Math.max(9, sheet.getLastColumn())).getValues();
      for (var r = 0; r < dataRange.length; r++) {
        var existingSurname = (dataRange[r][1] || '').toString().trim();
        var existingEmail = (dataRange[r][5] || '').toString().trim();
        var rowKey = (existingSurname + '||' + existingEmail).toLowerCase();
        if (rowKey === key) {
          targetRow = r + 2;
          break;
        }
      }
    }

    // Force the target mobile cell to text before writing the row.
    try { sheet.getRange(targetRow, 5).setNumberFormat('@'); } catch (_) {}
    sheet.getRange(targetRow, 1, 1, rowValues.length).setValues([rowValues]);
  } catch (err) {
    logError_(rallyName, 'handleEntrySubmit_', err, {
      rallyName: rallyName,
      formId: formId,
      formTitle: formTitle
    });
  }
}


/**
 * Ensure one onFormSubmit trigger for this exact entrant Form.
 * Other rallies' Entries triggers are preserved.
 */
function ensureEntrySubmitTrigger_(entrantForm) {
  var rallyName = '';
  var formId = '';
  try {
    if (!entrantForm) throw new Error('Entrant Form was not supplied');

/**

 *   rebuildEntriesFromResponses_('Crankhandle Trial');
 * Existing rows are rebuilt using Driver surname + Driver email; latest response wins.
 */
function rebuildEntriesFromResponses_(rallyName) {
  rallyName = normalizeRallyName_(rallyName);
  if (!rallyName) throw new Error('A rally name is required');

  try {
    var ssAdmin = findRallySpreadsheet_(rallyName);
    if (!ssAdmin) throw new Error('Rally spreadsheet not found');

    var destKey = 'ENTRIES_RESP_SSID__' + slugRallyName_(rallyName);

    if (!responsesId) throw new Error('Entries responses spreadsheet ID is not stored');

    var responsesSS = SpreadsheetApp.openById(responsesId);
    var raw = responsesSS.getSheetByName('Entries Responses');
    if (!raw) {
      var sheets = responsesSS.getSheets();
      for (var si = 0; si < sheets.length; si++) {
        if (/^Form[\s_]*responses[\s_]*\d+$/i.test(sheets[si].getName())) { raw = sheets[si]; break; }
      }
    }
    if (!raw || raw.getLastRow() < 2) throw new Error('No stored Entries responses were found');

    var values = raw.getDataRange().getDisplayValues();
    var headers = values[0].map(function(v) { return String(v || '').trim(); });
    var idx = {};
    for (var h = 0; h < headers.length; h++) idx[headers[h]] = h;

    var required = ['Driver first name','Driver surname','Navigator first name','Navigator surname','Driver mobile number','Driver email','Car make/model','Car year'];
    for (var q = 0; q < required.length; q++) {
      if (idx[required[q]] == null) throw new Error('Missing response column: ' + required[q]);
    }

    var latestByKey = {};
    var order = [];
    for (var r = 1; r < values.length; r++) {
      var src = values[r];
      var surname = String(src[idx['Driver surname']] || '').trim();
      var email = String(src[idx['Driver email']] || '').trim();
      if (!surname || !email) continue;
      var key = (surname + '||' + email).toLowerCase();
      if (!Object.prototype.hasOwnProperty.call(latestByKey, key)) order.push(key);
      latestByKey[key] = [
        String(src[idx['Driver first name']] || '').trim(),
        surname,
        String(src[idx['Navigator first name']] || '').trim(),
        String(src[idx['Navigator surname']] || '').trim(),
        String(src[idx['Driver mobile number']] || '').trim(),
        email,
        String(src[idx['Car make/model']] || '').trim(),
        String(src[idx['Car year']] || '').trim(),
        idx['Speed group'] == null ? '' : String(src[idx['Speed group']] || '').trim()
      ];
    }

    var sheet = ssAdmin.getSheetByName('Admin - Entries');
    if (!sheet) sheet = ssAdmin.insertSheet('Admin - Entries');

      'Driver first name','Driver surname','Navigator first name','Navigator surname',
      'Driver mobile number','Driver email','Car make/model','Car year','SG'
    ]]);
    try { sheet.getRange('E:E').setNumberFormat('@'); } catch (_) {}

    var rows = order.map(function(k) { return latestByKey[k]; });
    if (rows.length) sheet.getRange(2, 1, rows.length, 9).setValues(rows);
    return { ok: true, rallyName: rallyName, rebuiltRows: rows.length, sourceRows: values.length - 1 };
  } catch (err) {
    logError_(rallyName, 'rebuildEntriesFromResponses_', err, { rallyName: rallyName });
    throw err;
  }
}

/**
 * Parse a flexible STC time entry into a normalised HH:MM:SS string.
 * Accepts separators (: . ,) or packed digits (MMSS, HMMSS, HHMMSS).
 * Returns null if parsing fails.
 */
function parseStcTimeToHms_(raw, maxHour) {
  if (raw == null) return null;
  var s = String(raw).trim();
  if (!s) return null;

  // If contains separators, normalise . and , to :
  var hasSep = (s.indexOf(':') !== -1) || (s.indexOf('.') !== -1) || (s.indexOf(',') !== -1);
  var parts, h = 0, m = 0, sec = 0;

  if (hasSep) {
    s = s.replace(/[.,]/g, ':');
    parts = s.split(':');
    if (parts.length === 1) {
      // mm -> 00:mm:00
      h = 0;
      m = Number(parts[0]);
      sec = 0;
    } else if (parts.length === 2) {
      // mm:ss -> 00:mm:ss
      h = 0;
      m = Number(parts[0]);
      sec = Number(parts[1]);
    } else if (parts.length === 3) {
      h = Number(parts[0]);
      m = Number(parts[1]);
      sec = Number(parts[2]);
    } else {
      return null;
    }
  } else {
    // Packed digits: allow 3–6 digits
    if (!/^[0-9]+$/.test(s)) return null;
    if (s.length < 3 || s.length > 6) return null;

    if (s.length === 3) {
      // MSS -> 00:0M:SS
      h = 0;
      m = Number(s.substr(0, 1));
      sec = Number(s.substr(1, 2));
    } else if (s.length === 4) {
      // MMSS -> 00:MM:SS
      h = 0;
      m = Number(s.substr(0, 2));
      sec = Number(s.substr(2, 2));
    } else if (s.length === 5) {
      // HMMSS -> H:MM:SS
      h = Number(s.substr(0, 1));
      m = Number(s.substr(1, 2));
      sec = Number(s.substr(3, 2));
    } else {
      // 6 digits: HHMMSS
      h = Number(s.substr(0, 2));
      m = Number(s.substr(2, 2));
      sec = Number(s.substr(4, 2));
    }
  }

  if (!isFinite(h) || !isFinite(m) || !isFinite(sec)) return null;
  var maxH = (maxHour == null) ? 9 : Number(maxHour);
  if (!isFinite(maxH)) maxH = 9;
  if (h < 0 || h > maxH) return null; // elapsed STCs <= 9h; Start STC clock values may allow 23h
  if (m < 0 || m > 59 || sec < 0 || sec > 59) return null;

  // Format as HH:MM:SS
  var hh = (h < 10 ? '0' + h : String(h));
  var mm = (m < 10 ? '0' + m : String(m));
  var ss = (sec < 10 ? '0' + sec : String(sec));
  return hh + ':' + mm + ':' + ss;
}

/**
 * Google Form pathway only. Accept digits only:
 *   1–2 digits = seconds
 *   3–4 digits = minutes + seconds
 *   5–6 digits = hours + minutes + seconds
 * Clock hours up to 23 are allowed because Marshal/Start-STC form values can be
 * rally master-clock times. The STC Recorder/PWA continues to use parseStcTimeToHms_.
 */
function parseFormDigitsTimeToHms_(raw) {
  if (raw == null) return null;
  var s = String(raw).trim();
  if (!/^[0-9]{1,6}$/.test(s)) return null;

  var h = 0, m = 0, sec = 0;
  if (s.length <= 2) {
    sec = Number(s);
  } else if (s.length <= 4) {
    m = Number(s.substring(0, s.length - 2));
    sec = Number(s.substring(s.length - 2));
  } else {
    h = Number(s.substring(0, s.length - 4));
    m = Number(s.substring(s.length - 4, s.length - 2));
    sec = Number(s.substring(s.length - 2));
  }

  if (!isFinite(h) || !isFinite(m) || !isFinite(sec)) return null;
  if (h < 0 || h > 23 || m < 0 || m > 59 || sec < 0 || sec > 59) return null;

  var hh = h < 10 ? '0' + h : String(h);
  var mm = m < 10 ? '0' + m : String(m);
  var ss = sec < 10 ? '0' + sec : String(sec);
  return hh + ':' + mm + ':' + ss;
}

/**
 * handleStcSubmit_Day1_
 * Handler for STC Day 1 submissions.
 * - Triggered only for the Day 1 STC form (ensureStcSubmitTriggerForDay1_)
 * - Converts digits-only form inputs to HH:MM:SS using parseFormDigitsTimeToHms_
 * - Finds the entrant row by rally number + surname (case-insensitive)
 *   and overwrites that row's STC columns (no new rows are added).
 */
function handleStcSubmit_Day1_(e) {
  try {
    var form = e && e.source;
    if (!form || typeof form.getDestinationId !== 'function') return;

    var destId = form.getDestinationId && form.getDestinationId();
    if (!destId) return;

    var ss = SpreadsheetApp.openById(destId);
    var sheetName = 'STC Day 1';
    var sheet = ss.getSheetByName(sheetName);
    if (!sheet) {
      // Skeleton sheets should already exist from uploadControls; if not, fail safely.
      logError_('handleStcSubmit_Day1_', 'Missing sheet "' + sheetName + '" in destination spreadsheet');
      return;
    }

    var itemResponses = (e && e.response && e.response.getItemResponses)
      ? e.response.getItemResponses()
      : [];
    var byTitle = {};
    for (var i = 0; i < itemResponses.length; i++) {
      var ir = itemResponses[i];
      var title = ir.getItem().getTitle();
      byTitle[title] = ir.getResponse();
    }

    var rallyNumber = (byTitle['Rally Number'] || '').toString().trim();
    var driverSurname = (byTitle['Driver Surname'] || '').toString().trim();

    // Require both rally number and surname to be present.
    if (!rallyNumber || !driverSurname) {
      return;
    }

    // Collect STC fields in the order they appear on the form and normalise times.
// Titles may be "STC 1", "Start STC 1", etc. We preserve form order and write sequentially into STC columns.
    var stcValues = [];
    var rawStcValues = [];
    var stcLabels = [];
    var stcIndex = 1;
    for (var iR = 0; iR < itemResponses.length; iR++) {
      var ir2 = itemResponses[iR];
      var title2 = ir2.getItem().getTitle();
      var t2 = String(title2 || '').trim();
      // Accept any STC-family field (e.g., "STC 1", "Start STC 1").
      if (!/^((Start\s+)?STC)\s+\d+$/i.test(t2)) continue;

      var rawVal = (ir2.getResponse() || '').toString().trim();
      var normalised = parseFormDigitsTimeToHms_(rawVal);
      rawStcValues.push(rawVal);

      if (rawVal !== '' && normalised == null) {
        logStcNoMatch_(ss, dayVal, rallyNumber, driverSurname, rawStcValues, 'INVALID_TIME_STC' + stcIndex);
        return;
      }

      stcValues.push(normalised != null ? normalised : '');
      stcLabels.push(t2.replace(/\s+Time$/i, '').trim());
      stcIndex++;
    }

    if (stcValues.length === 0) {
      // Nothing to write.
      return;
    }

    // Find matching entrant row in STC Day 1 by rally number + surname (case-insensitive).
    var surnameKey = driverSurname.toString().trim().toUpperCase();
    var lastRow = sheet.getLastRow();
    if (lastRow < 2) {
      // No entrant rows.
      return;
    }
    var range = sheet.getRange(2, 1, lastRow - 1, 2); // Entrant, #
    var values = range.getValues();
    var targetRowIndex = -1; // 1-based row index in sheet

    for (var r = 0; r < values.length; r++) {
      var nameCell = (values[r][0] != null) ? String(values[r][0]).trim() : '';
      var rallyCell = (values[r][1] != null) ? String(values[r][1]).trim() : '';
      if (!rallyCell) continue;
      if (rallyCell === rallyNumber && nameCell.toUpperCase() === surnameKey) {
        targetRowIndex = 2 + r;
        break;
      }
    }

    if (targetRowIndex === -1) {
      // No matching entrant for this rally number + surname; log to STC Errors and do not modify the main sheet.
      logStcNoMatch_(ss, dayVal, rallyNumber, driverSurname, stcValues, 'NO_MATCH_RALLY_SURNAME');
      return;
    }

    // v3.01: legacy Day 1 handler also writes by STC index into Manual side columns.
    // This prevents sequential writes from filling GPS columns when a Google Form is used.
    var dayVal = 1;
    var side = 'M';
    var headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getDisplayValues()[0] || [];
    for (var sv = 0; sv < stcValues.length; sv++) {
      var targetCol = -1;

      // Google Form submissions can contain Start STC plus ordinary STCs whose
      // numbering is not sequential by array position. When labels are supplied,
      // write to the exact matching control column first.
      if (Array.isArray(stcLabels) && stcLabels[sv]) {
        targetCol = stcTargetColumnForLabelSide_(headers, stcLabels[sv], side);
      }

      // Keep the existing index-based behaviour as the compatibility fallback
      // for legacy callers/PWA submissions that do not supply form labels.
      if (targetCol < 1) targetCol = stcTargetColumnForSide_(headers, sv, side);

      sheet.getRange(targetRowIndex, targetCol).setValue(stcValues[sv] || '');
    }
  } catch (err) {
    logError_('handleStcSubmit_Day1_', err);
  }
}

function handleMarshalSubmit_Day1_(e) {
  try {
    var form = e && e.source;
    if (!form || typeof form.getDestinationId !== 'function') return;

    var destId = form.getDestinationId && form.getDestinationId();
    if (!destId) return;

    var ss = SpreadsheetApp.openById(destId);
    var sheetName = 'Marshal Day 1';
    var sheet = ss.getSheetByName(sheetName);
    if (!sheet) {
      logError_('handleMarshalSubmit_Day1_', 'Missing sheet "' + sheetName + '" in destination spreadsheet');
      return;
    }

    var itemResponses = (e && e.response && e.response.getItemResponses)
      ? e.response.getItemResponses()
      : [];
    var byTitle = {};
    for (var i = 0; i < itemResponses.length; i++) {
      var ir = itemResponses[i];
      var title = ir.getItem().getTitle();
      byTitle[title] = ir.getResponse();
    }

    var marshalNumberRaw = (byTitle['Control Number'] || byTitle['Marshal Number'] || '').toString().trim();
    if (!marshalNumberRaw) {
      // No marshal identifier; nothing to map.
      return;
    }
        var key = marshalNumberRaw.toUpperCase();
    var wantedHeader = '';
    if (/^OC\d+$/.test(key)) {
      wantedHeader = 'Open Control ' + key.replace(/^OC/, '');
    } else if (/^\d+$/.test(key)) {
      wantedHeader = 'Marshal ' + key;
    }
    var targetCol = 0;
    try {
      var headerRow = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0] || [];
      for (var hc = 0; hc < headerRow.length; hc++) {
        if (String(headerRow[hc]).trim() === wantedHeader) { targetCol = hc + 1; break; }
      }
    } catch (e2) {}
    // Legacy fallback: Column 1 is '#', so Marshal 1 is column 2, Marshal 2 is column 3, etc.
    if (!targetCol) {
      var marshalColIndex = parseInt(marshalNumberRaw, 10);
      if (!isFinite(marshalColIndex) || marshalColIndex <= 0) { return; }
      targetCol = 1 + marshalColIndex;
    }
// Collect Car N Time fields in numeric order (Car 1 Time, Car 2 Time, ...)
    // and normalise times using the same helper as STC.
    var carTimes = [];
    var carNo = 1;
    while (true) {

      if (!(key in byTitle)) break;

      var normalised = parseFormDigitsTimeToHms_(rawVal);
      carTimes.push(normalised != null ? normalised : rawVal);
      carNo++;
    }

    if (carTimes.length === 0) {
      // Nothing to write.
      return;
    }

    // Ensure the '#' column has car numbers 1..N for the rows we are about to touch.
    var numCars = carTimes.length;
    var hashRange = sheet.getRange(2, 1, numCars, 1);
    var hashValues = hashRange.getValues();
    for (var r = 0; r < numCars; r++) {
      var current = hashValues[r][0];
      if (current === '' || current == null) {
        hashValues[r][0] = r + 1; // Car number (1-based)
      }
    }
    hashRange.setValues(hashValues);

    // Write car times vertically into the Marshal N column, rows 2..(1+numCars).
    var colRange = sheet.getRange(2, targetCol, numCars, 1);
    var colValues = [];
    for (var r2 = 0; r2 < numCars; r2++) {
      colValues.push([carTimes[r2] || '']);
    }
    colRange.setValues(colValues);
  } catch (err) {
    logError_('handleMarshalSubmit_Day1_', err);
  }
}

function parseRallyNameFromFormTitle_(title) {
  // Support both rally-first and rally-last naming, and both dash styles.
  // Examples we accept:
  //   "<Rally Title> – STC Times – Day X"
  //   "<Rally Title> – STC – Day X"
  //   "STC – Day X – <Rally Title>"
  //   "<Rally Title> – Marshal – Day X"
  //   "Marshal – Day X – <Rally Title>"
  //
  // NOTE: We return the rally title segment only (trimmed). If no pattern matches,
  // we fall back to the previous heuristic.

  var t = String(title || '').trim();
  if (!t) return '';

  // Normalise common dash variants for parsing (keeps original string intact).
  var tn = t.replace(/\s-\s/g, ' – ');

  // 1) Rally-first STC Times
  var m = tn.match(/^(.*?)\s–\sSTC\sTimes\s–\sDay\s(\d+)\b/i);
  if (m && m[1]) return String(m[1]).trim();

  // 2) Rally-first STC
  m = tn.match(/^(.*?)\s–\sSTC\s–\sDay\s(\d+)\b/i);
  if (m && m[1]) return String(m[1]).trim();

  // 3) STC-first
  m = tn.match(/^STC(?:\sTimes)?\s–\sDay\s(\d+)\s–\s(.+)$/i);
  if (m && m[2]) return String(m[2]).trim();

  // 4) Rally-first Marshal
  m = tn.match(/^(.*?)\s–\sMarshal\s–\sDay\s(\d+)\b/i);
  if (m && m[1]) return String(m[1]).trim();

  // 5) Marshal-first
  m = tn.match(/^Marshal\s–\sDay\s(\d+)\s–\s(.+)$/i);
  if (m && m[2]) return String(m[2]).trim();

  // Fallback: take everything before the last " – Day " (or " - Day ")
  var j = tn.lastIndexOf(' – Day ');
  if (j > 0) return tn.substring(0, j).trim();

  // Last resort (keeps legacy behaviour)
  return t;
}

  }

function handleStcSubmit_AllDays_(e) {
  try {
    var form = e && e.source;
    if (!form || typeof form.getTitle !== 'function') return;

    var tag = getFormTag_(form.getId());
    if (!tag || !tag.kind || String(tag.kind) !== 'STC') {
      // STC trigger should only be attached to STC forms; do nothing if not.
      return;
    }

    // Watermark-first: do NOT rely on the visible form title for routing.
    var dayVal = tag.day ? Number(tag.day) : null;
    var rallyFolderId = tag.rallyFolderId ? String(tag.rallyFolderId) : '';
    var rallyName = tag.rallyName ? String(tag.rallyName) : '';

    if (!rallyName && rallyFolderId) {
      rallyName = deriveRallyNameFromFolderId_(rallyFolderId);
    }

    // If we still cannot resolve the rally context, log to STC Errors if possible (no silent exits).
    if (!dayVal || !rallyName) {
      var ssFallback = rallyFolderId ? openControlsSpreadsheetFromRallyFolderId_(rallyFolderId) : null;
      if (ssFallback) {
        logStcNoMatch_(ssFallback, dayVal || '', '', '', [], 'MISSING_WATERMARK_FIELDS');
      }
      return;
    }

    var storage = ensureRallyStorage_(rallyName);
    var ss = storage && storage.spreadsheet;
    if (!ss) return;

    // Exact 03:00 safety backstop: even if Google's clock trigger is delayed,
    // a stale form cannot write a response outside its rally-day window.
    var submittedAtGuard = (e && e.response && e.response.getTimestamp) ? e.response.getTimestamp() : new Date();
    if (tag.dateLabel) {
      var stcWindow = rallyDayWindow_(tag.dateLabel);
      if (stcWindow && (submittedAtGuard.getTime() < stcWindow.start.getTime() || submittedAtGuard.getTime() >= stcWindow.end.getTime())) {
        logStcNoMatch_(ss, dayVal, '', '', [], 'FORM_OUTSIDE_DAY_WINDOW');
        return;
      }
    }

    var sheetName = 'STC Day ' + dayVal;
    var sheet = ss.getSheetByName(sheetName);
    if (!sheet) {
      logError_('handleStcSubmit_AllDays_', 'Missing sheet "' + sheetName + '" in rally spreadsheet');
      return;
    }

    var itemResponses = (e && e.response && e.response.getItemResponses)
      ? e.response.getItemResponses()
      : [];
    var byTitle = {};
    for (var i = 0; i < itemResponses.length; i++) {
      var ir = itemResponses[i];
      var it = ir.getItem && ir.getItem();
      var t = it && it.getTitle ? it.getTitle() : '';
      byTitle[t] = ir.getResponse();
    }

    var rallyNumber = (byTitle['Rally Number'] || '').toString().trim();

    // Surname is required. Prefer the explicit "Driver Surname" field, but allow common aliases.
    var driverSurname = (byTitle['Driver Surname'] || byTitle['Entrant'] || byTitle['Surname'] || '').toString().trim();

    if (!rallyNumber || !driverSurname) {
      logStcNoMatch_(ss, dayVal, rallyNumber || '', driverSurname || '', [], 'MISSING_KEYS');
      return;
    }

    var stcValues = [];
    var rawStcValues = [];
    var stcLabels = [];

    var stcIndex = 1;
    for (var iR = 0; iR < itemResponses.length; iR++) {
      var ir2 = itemResponses[iR];
      var title2 = ir2.getItem().getTitle();
      var t2 = String(title2 || '').trim();
      // Accept STC-family fields, including legacy "... Time".
      if (!/^((Start\s+)?STC)\s+\d+(\s+Time)?$/i.test(t2)) continue;

      var rawVal = (ir2.getResponse() || '').toString().trim();
      var normalised = parseFormDigitsTimeToHms_(rawVal);
      rawStcValues.push(rawVal);

      if (rawVal !== '' && normalised == null) {
        logStcNoMatch_(ss, dayVal, rallyNumber, driverSurname, rawStcValues, 'INVALID_TIME_STC' + stcIndex);
        return;
      }

      stcValues.push(normalised != null ? normalised : '');
      stcLabels.push(t2.replace(/\s+Time$/i, '').trim());
      stcIndex++;
    }

    if (stcValues.length === 0) return;

    var submittedAt = (e && e.response && e.response.getTimestamp) ? e.response.getTimestamp() : new Date();
    applyStcSubmissionToDaySheet_(ss, dayVal, rallyNumber, driverSurname, stcValues, rawStcValues, 'FORM', submittedAt, 'M', stcLabels);
  } catch (err) {
    logError_('handleStcSubmit_AllDays_', err);
  }
}

function readStcLabelsForDay_(ss, dayVal) {
  var out = [];
  if (!ss) return out;
  try {
    var day = String(dayVal || '1');
    var cfg = ss.getSheetByName(CONTROLS_SHEET_NAME);
    if (cfg && cfg.getLastRow() >= 2) {
      var vals = cfg.getRange(1, 1, cfg.getLastRow(), cfg.getLastColumn()).getDisplayValues();
      var head = vals[0] || [];
      function col_(name, fallback) {
        for (var i = 0; i < head.length; i++) {
          if (String(head[i]).trim().toLowerCase() === String(name).trim().toLowerCase()) return i;
        }
        return fallback;
      }
      var dayCol = col_('Day', 0);
      var typeCol = col_('Type', 2);
      var labelCol = col_('Label', 3);
      var enabledCol = col_('Enabled', 4);
      for (var r = 1; r < vals.length; r++) {
        var row = vals[r];
        if (String(row[dayCol] || '') !== day) continue;
        if (String(row[typeCol] || '').trim().toLowerCase() !== 'stc') continue;
        var enabledText = String(row[enabledCol] || '').trim().toLowerCase();
        var enabled = (enabledText === 'true' || enabledText === 'yes' || enabledText === '1' || enabledText === 'enabled');
        if (!enabled) continue;
        out.push(String(row[labelCol] || ('STC ' + (out.length + 1))).trim());
      }
      if (out.length) return out;
    }

    var daySheet = ss.getSheetByName('STC Day ' + day);
    if (daySheet && daySheet.getLastColumn() >= 3) {
      var headers = daySheet.getRange(1, 3, 1, daySheet.getLastColumn() - 2).getDisplayValues()[0] || [];
      var seen = {};
      headers.forEach(function(h, idx) {
        var label = String(h || ('STC ' + (idx + 1))).trim();
        label = label.replace(/\s+(M|Manual|GPS)$/i, '').trim();
        if (label && !seen[label]) { seen[label] = true; out.push(label); }
      });
    }
  } catch (_) {}
  return out;
}

function normalizeStcSubmitSide_(sideOrMode, fallbackMode) {
  var s = String(sideOrMode || '').trim().toLowerCase();
  if (s === 'gps' || s === 'g') return 'GPS';
  if (s === 'manual' || s === 'm' || s === 'form') return 'M';
  return normalizeTimingMode_(fallbackMode) === 'GPS' ? 'GPS' : 'M';
}

function stcTargetColumnForSide_(headers, stcIndexZero, side) {
  var idx = Number(stcIndexZero || 0);
  var suffix = (side === 'GPS') ? 'gps' : 'm';
  var baseN = idx + 1;

  // Preferred v3.00 paired columns: "STC n M" and "STC n GPS".
  for (var h = 2; h < headers.length; h++) {
    var raw = String(headers[h] || '').trim();
    var norm = raw.toLowerCase().replace(/\s+/g, ' ');
    var m = norm.match(/^(?:start\s+)?stc\s+(\d+)\s+(m|manual|gps)$/i);
    if (m && parseInt(m[1], 10) === baseN) {
      var found = String(m[2] || '').toLowerCase();
      if ((suffix === 'gps' && found === 'gps') || (suffix === 'm' && (found === 'm' || found === 'manual'))) {
        return h + 1; // 1-based
      }
    }
  }

  // Structural fallback for v3.00 paired sheets.
  var pairedCol = 3 + (idx * 2) + (side === 'GPS' ? 1 : 0);
  if (headers.length >= pairedCol) return pairedCol;

  // Back-compat fallback for older one-column sheets.
  return 3 + idx;
}

function stcTargetColumnForLabelSide_(headers, controlLabel, side) {
  var wanted = String(controlLabel || '').trim().toLowerCase().replace(/\s+/g, ' ');
  if (!wanted) return -1;
  var suffixes = side === 'GPS' ? [' gps'] : [' m', ' manual'];

  for (var h = 2; h < headers.length; h++) {
    var norm = String(headers[h] || '').trim().toLowerCase().replace(/\s+/g, ' ');
    for (var s = 0; s < suffixes.length; s++) {
      if (norm === wanted + suffixes[s]) return h + 1; // 1-based
    }
  }
  return -1;
}

function applyStcSubmissionToDaySheet_(ss, dayVal, rallyNumber, driverSurname, stcValues, rawStcValues, source, submittedAt, submitSide, stcLabels) {
  var lock = null;
  try {
    lock = LockService.getScriptLock();
    lock.waitLock(30000);
  } catch (_) {
    lock = null;
  }

  try {
    if (!ss) return { ok: false, error: 'Missing spreadsheet' };
    var sheetName = 'STC Day ' + String(dayVal || '1');
    var sheet = ss.getSheetByName(sheetName);
    if (!sheet) {
      logError_('applyStcSubmissionToDaySheet_', 'Missing sheet "' + sheetName + '" in rally spreadsheet');
      return { ok: false, error: 'Missing sheet: ' + sheetName };
    }
    if (!rallyNumber || !driverSurname) {
      logStcNoMatch_(ss, dayVal, rallyNumber || '', driverSurname || '', rawStcValues || stcValues || [], 'MISSING_KEYS_' + (source || 'STC'));
      return { ok: false, error: 'Missing rally number or surname' };
    }
    if (!stcValues || !stcValues.length) return { ok: false, error: 'No STC values supplied' };

    var submitDate = normaliseStcSubmitTimestamp_(submittedAt);
    var surnameKey = String(driverSurname).trim().toUpperCase();
    var rallyKey = String(rallyNumber).trim();
    var lastRow = sheet.getLastRow();
    if (lastRow < 2) return { ok: false, error: 'No entrant rows' };

    var values = sheet.getRange(2, 1, lastRow - 1, 2).getValues();
    var targetRowIndex = -1;
    for (var r = 0; r < values.length; r++) {
      var nameCell = (values[r][0] != null) ? String(values[r][0]).trim() : '';
      var rallyCell = (values[r][1] != null) ? String(values[r][1]).trim() : '';
      if (!rallyCell) continue;
      if (rallyCell === rallyKey && nameCell.toUpperCase() === surnameKey) {
        targetRowIndex = 2 + r;
        break;
      }
    }
    if (targetRowIndex === -1) {
      logStcNoMatch_(ss, dayVal, rallyNumber, driverSurname, rawStcValues || stcValues, 'NO_MATCH_RALLY_SURNAME_' + (source || 'STC'));
      return { ok: false, error: 'No matching entrant' };
    }

    var metaSheet = ensureStcLatestMetaSheet_(ss);
    var meta = findStcLatestMetaRow_(metaSheet, dayVal, rallyNumber, driverSurname);
    if (meta && meta.row > 0 && meta.timestamp && meta.timestamp.getTime && submitDate.getTime() < meta.timestamp.getTime()) {
      return {
        ok: true,
        skipped: true,
        reason: 'OLDER_THAN_CURRENT',
        sheetName: sheetName,
        row: targetRowIndex,
        source: source || 'STC',
        submittedAt: submitDate.toISOString(),
        currentLatestAt: meta.timestamp.toISOString()
      };
    }

    // v3.01: STC Day sheets now have paired columns for each control:
    //   STC 1 M | STC 1 GPS | STC 2 M | STC 2 GPS | ...
    // Do not write a compact block from column 3, because that shifts STC 2
    // into the STC 1 GPS column. Resolve each STC index to its target side.
    var timingModeForDay = getTimingModeForDay_(ss, dayVal);
    var side = normalizeStcSubmitSide_(submitSide || source, timingModeForDay);
    var headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getDisplayValues()[0] || [];
    for (var sv = 0; sv < stcValues.length; sv++) {
      var targetCol = -1;

      // Google Form submissions carry the real control labels. This matters when
      // a Start STC is present because array position no longer equals STC number.
      if (Array.isArray(stcLabels) && stcLabels[sv]) {
        targetCol = stcTargetColumnForLabelSide_(headers, stcLabels[sv], side);
      }

      // Compatibility fallback for PWA/legacy callers that do not pass labels.
      if (targetCol < 1) targetCol = stcTargetColumnForSide_(headers, sv, side);

      sheet.getRange(targetRowIndex, targetCol).setValue(stcValues[sv] || '');
    }

    if (metaSheet) {
      var metaRow = meta && meta.row > 0 ? meta.row : Math.max(2, metaSheet.getLastRow() + 1);
      metaSheet.getRange(metaRow, 1, 1, 7).setValues([[
        String(dayVal || ''),
        String(rallyNumber || ''),
        String(driverSurname || ''),
        submitDate,
        String(source || 'STC'),
        stcValues.length,
        new Date()
      ]]);
      try { metaSheet.hideSheet(); } catch (_) {}
    }

    return {
      ok: true,
      sheetName: sheetName,
      row: targetRowIndex,
      columnsWritten: stcValues.length,
      source: source || 'STC',
      sourceSide: side,
      submittedAt: submitDate.toISOString()
    };
  } finally {
    try { if (lock) lock.releaseLock(); } catch (_) {}
  }
}


function normalisePwaGpsQualityLetter_(quality) {
  var q = String(quality || '').trim().toUpperCase();
  if (!q) return '';
  if (q === 'GOOD' || q === 'G') return 'G';
  if (q === 'FAIR' || q === 'F') return 'F';
  if (q === 'POOR' || q === 'P') return 'P';
  if (q === 'NO GPS' || q === 'NO_GPS' || q === 'NONE' || q === 'N') return 'N';
  return q.charAt(0);
}

function buildPwaGpsSheetValue_(gpsElapsed, gpsQuality, maxHour) {
  var quality = normalisePwaGpsQualityLetter_(gpsQuality);
  var rawTime = String(gpsElapsed || '').trim();

  var time = parseStcTimeToHms_(rawTime, maxHour);
  if (time == null) return null;
  return quality ? (time + ' ' + quality) : time;
}

      logError_('applyStcPwaPairedSubmissionToDaySheet_', 'Missing sheet "' + sheetName + '" in rally spreadsheet');
      return { ok: false, error: 'Missing sheet: ' + sheetName };
    }
    if (!rallyNumber || !driverSurname) {
      logStcNoMatch_(ss, dayVal, rallyNumber || '', driverSurname || '', rawStcPairs || stcPairs || [], 'MISSING_KEYS_PWA');
      return { ok: false, error: 'Missing rally number or surname' };
    }
    if (!stcPairs || !stcPairs.length) return { ok: false, error: 'No STC values supplied' };

    var submitDate = normaliseStcSubmitTimestamp_(submittedAt);
    var surnameKey = String(driverSurname).trim().toUpperCase();
    var rallyKey = String(rallyNumber).trim();
    var lastRow = sheet.getLastRow();
    if (lastRow < 2) return { ok: false, error: 'No entrant rows' };

    var values = sheet.getRange(2, 1, lastRow - 1, 2).getValues();
    var targetRowIndex = -1;
    for (var r = 0; r < values.length; r++) {
      var nameCell = (values[r][0] != null) ? String(values[r][0]).trim() : '';
      var rallyCell = (values[r][1] != null) ? String(values[r][1]).trim() : '';
      if (!rallyCell) continue;
      if (rallyCell === rallyKey && nameCell.toUpperCase() === surnameKey) {
        targetRowIndex = 2 + r;
        break;
      }
    }
    if (targetRowIndex === -1) {
      logStcNoMatch_(ss, dayVal, rallyNumber, driverSurname, rawStcPairs || stcPairs, 'NO_MATCH_RALLY_SURNAME_PWA');
      return { ok: false, error: 'No matching entrant' };
    }

    var metaSheet = ensureStcLatestMetaSheet_(ss);
    var meta = findStcLatestMetaRow_(metaSheet, dayVal, rallyNumber, driverSurname);
    if (meta && meta.row > 0 && meta.timestamp && meta.timestamp.getTime && submitDate.getTime() < meta.timestamp.getTime()) {
      return {
        ok: true,
        skipped: true,
        reason: 'OLDER_THAN_CURRENT',
        sheetName: sheetName,
        row: targetRowIndex,
        source: 'PWA',
        submittedAt: submitDate.toISOString(),
        currentLatestAt: meta.timestamp.toISOString()
      };
    }

    var headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getDisplayValues()[0] || [];
    var columnsWritten = 0;
    for (var sv = 0; sv < stcPairs.length; sv++) {
      var pair = stcPairs[sv] || {};
      var pairLabel = String(pair.label || '').trim();
      var manualCol = stcTargetColumnForLabelSide_(headers, pairLabel, 'M');
      var gpsCol = stcTargetColumnForLabelSide_(headers, pairLabel, 'GPS');
      if (manualCol < 1) manualCol = stcTargetColumnForSide_(headers, sv, 'M');
      if (gpsCol < 1) gpsCol = stcTargetColumnForSide_(headers, sv, 'GPS');
      sheet.getRange(targetRowIndex, manualCol).setValue(pair.button || '');
      sheet.getRange(targetRowIndex, gpsCol).setValue(pair.gps || '');
      columnsWritten += 2;
    }

    if (metaSheet) {
      var metaRow = meta && meta.row > 0 ? meta.row : Math.max(2, metaSheet.getLastRow() + 1);
      metaSheet.getRange(metaRow, 1, 1, 7).setValues([[
        String(dayVal || ''),
        String(rallyNumber || ''),
        String(driverSurname || ''),
        submitDate,
        'PWA',
        stcPairs.length,
        new Date()
      ]]);
      try { metaSheet.hideSheet(); } catch (_) {}
    }

    return {
      ok: true,
      sheetName: sheetName,
      row: targetRowIndex,
      columnsWritten: columnsWritten,
      source: 'PWA',
      sourceSide: 'M+GPS',
      submittedAt: submitDate.toISOString()
    };
  } finally {
    try { if (lock) lock.releaseLock(); } catch (_) {}
  }
}

function stcPwaSubmit_(options) {
  var rallyName = (options && options.rallyName) || 'Rally';
  var dayVal = String((options && options.day) || '1');
  var payload = {};
  try {
    var raw = (options && options.data) || '{}';
    payload = (typeof raw === 'string') ? JSON.parse(raw) : (raw || {});
  } catch (err) {
    return { ok: false, mode: 'stcPwaSubmit', error: 'Invalid JSON payload' };
  }
  if (payload.rallyName || payload.rallyTitle) rallyName = normalizeRallyName_(payload.rallyName || payload.rallyTitle);
  if (payload.day) dayVal = String(payload.day);
  if (!payload.day && payload.dayLabel) {
    var dm = String(payload.dayLabel).match(/(\d+)/);
    if (dm) dayVal = dm[1];
  }
  if (!isValidRallyName_(rallyName)) return { ok: false, mode: 'stcPwaSubmit', error: 'Invalid rallyName' };
  try {
    // v3.02: PWA submit contract is structured and paired:
    //   - button/manual elapsed time -> STC n M
    //   - selected GPS elapsed time + quality letter -> STC n GPS, e.g. "00:12:33 G"
    // Start time remains local to the PWA and is not required by this submit pathway.
    var storage = rcRetry_('stcPwaSubmit_ ensureRallyStorage_', function () {
      return ensureRallyStorage_(rallyName);
    }, 4, 900);
    var ss = storage.spreadsheet;
    var rallyNumber = String(payload.rallyNo || payload.rallyNumber || '').trim();
    var driverSurname = String(payload.surname || payload.driverSurname || payload.name || '').trim();
    var stcs = Array.isArray(payload.stcs) ? payload.stcs : [];
    var stcLabels = [];
    var pairedMode = false;

    for (var has = 0; has < stcs.length; has++) {
      if (stcPwaItemHasPairedFields_(stcs[has] || {})) { pairedMode = true; break; }
    }

    if (pairedMode) {
      var stcPairs = [];
      var rawPairs = [];
      var rawAppendValues = [];

      for (var i = 0; i < stcs.length; i++) {
        var item = stcs[i] || {};
        var label = String(item.label || ('STC ' + (i + 1))).trim();
        stcLabels.push(label);

        var isStartStc = /^start\s+stc\s+\d+$/i.test(label);
        var rawManual = String(item.buttonElapsed || item.manualElapsed || item.buttonTime || item.manualTime || '').trim();
        var manualValue = '';
        if (rawManual) {
          manualValue = parseStcTimeToHms_(rawManual, isStartStc ? 23 : 9);
          if (manualValue == null) {
            logStcNoMatch_(ss, dayVal, rallyNumber, driverSurname, rawPairs, 'INVALID_TIME_PWA_' + label + '_M');
            return { ok: false, mode: 'stcPwaSubmit', error: 'Invalid button/manual time at ' + label };
          }
        }

        var rawGps = String(item.gpsElapsed || item.gpsTime || item.selectedGpsElapsed || item.selectedGpsTime || '').trim();
        var rawQuality = String(item.gpsQuality || item.gpsQ || item.quality || '').trim();
        var gpsValue = buildPwaGpsSheetValue_(rawGps, rawQuality, isStartStc ? 23 : 9);
        if (gpsValue == null) {
          logStcNoMatch_(ss, dayVal, rallyNumber, driverSurname, rawPairs, 'INVALID_TIME_PWA_' + label + '_GPS');
          return { ok: false, mode: 'stcPwaSubmit', error: 'Invalid GPS time at ' + label };
        }

        stcPairs.push({ label: label, button: manualValue || '', gps: gpsValue || '' });
        rawPairs.push({ label: label, button: rawManual || '', gps: gpsValue || '' });
        rawAppendValues.push(manualValue || '');
        rawAppendValues.push(gpsValue || '');
      }

      var pwaSs = ensureStcPwaResponsesWorkbook_(storage.rallyFolder, storage.rallyName, dayVal);
      var rawSheet = ensureStcPwaResponsesSheet_(pwaSs, dayVal, stcLabels);
      var submittedAt = new Date();
      if (rawSheet) rawSheet.appendRow([submittedAt, rallyNumber, driverSurname, 'PWA'].concat(rawAppendValues));
      cleanupLegacyStcPwaTabsFromMain_(ss);
      var applied = applyStcPwaPairedSubmissionToDaySheet_(ss, dayVal, rallyNumber, driverSurname, stcPairs, rawPairs, submittedAt);
      return Object.assign({
        ok: !!applied.ok,
        mode: 'stcPwaSubmit',
        rallyName: storage.rallyName,
        day: dayVal,
        contract: 'PWA_STC_M_PLUS_GPS_V3_02',
        rawSheetName: rawSheet ? rawSheet.getName() : '',
        rawSpreadsheetUrl: pwaSs ? pwaSs.getUrl() : ''
      }, applied);
    }

    // Backwards compatibility for older PWA payloads that submit one side only.
    var stcValues = [];
    var rawStcValues = [];
    for (var j = 0; j < stcs.length; j++) {
      var oldItem = stcs[j] || {};
      var oldLabel = String(oldItem.label || ('STC ' + (j + 1))).trim();
      var rawVal = String(oldItem.elapsed || oldItem.time || '').trim();
      var normalised = rawVal ? parseStcTimeToHms_(rawVal) : '';
      stcLabels.push(oldLabel);
      rawStcValues.push(rawVal);
      if (rawVal && normalised == null) {
        logStcNoMatch_(ss, dayVal, rallyNumber, driverSurname, rawStcValues, 'INVALID_TIME_PWA_STC' + (j + 1));
        return { ok: false, mode: 'stcPwaSubmit', error: 'Invalid time at ' + oldLabel };
      }
      stcValues.push(normalised || '');
    }
    var legacyPwaSs = ensureStcPwaResponsesWorkbook_(storage.rallyFolder, storage.rallyName, dayVal);
    var legacyRawSheet = ensureStcPwaResponsesSheet_(legacyPwaSs, dayVal, stcLabels);
    var legacySubmittedAt = new Date();
    if (legacyRawSheet) legacyRawSheet.appendRow([legacySubmittedAt, rallyNumber, driverSurname, 'PWA'].concat(rawStcValues));
    cleanupLegacyStcPwaTabsFromMain_(ss);
    var pwaSubmitSide = normalizeStcSubmitSide_(payload.source || payload.timingMode || payload.mode, getTimingModeForDay_(ss, dayVal));
    var legacyApplied = applyStcSubmissionToDaySheet_(ss, dayVal, rallyNumber, driverSurname, stcValues, rawStcValues, 'PWA', legacySubmittedAt, pwaSubmitSide);
    return Object.assign({ ok: !!legacyApplied.ok, mode: 'stcPwaSubmit', rallyName: storage.rallyName, day: dayVal, contract: 'LEGACY_SINGLE_SIDE', rawSheetName: legacyRawSheet ? legacyRawSheet.getName() : '', rawSpreadsheetUrl: legacyPwaSs ? legacyPwaSs.getUrl() : '' }, legacyApplied);
  } catch (err2) {
    logError_(rallyName, 'stcPwaSubmit_', err2, { day: dayVal, payload: payload });
    return { ok: false, mode: 'stcPwaSubmit', error: String(err2) };
  }
}

function handleMarshalSubmit_AllDays_(e) {
  try {
    var form = e && e.source;
    if (!form || typeof form.getTitle !== 'function') return;

    var title = form.getTitle();
    var tag = getFormTag_(form.getId());
    if (tag && tag.kind && String(tag.kind) !== 'MARSHAL') tag = null;
    var dayVal = (tag && tag.day) ? Number(tag.day) : parseDayFromFormTitle_(title);
    var rallyName = (tag && tag.rallyName) ? String(tag.rallyName) : parseRallyNameFromFormTitle_(title);
    if (!dayVal || !rallyName) return;

    var storage = ensureRallyStorage_(rallyName);
    var ss = storage && storage.spreadsheet;
    if (!ss) return;

    // Exact 03:00 safety backstop for stale/late-open Marshal forms.
    var submittedAtGuard = (e && e.response && e.response.getTimestamp) ? e.response.getTimestamp() : new Date();
    if (tag && tag.dateLabel) {
      var marshalWindow = rallyDayWindow_(tag.dateLabel);
      if (marshalWindow && (submittedAtGuard.getTime() < marshalWindow.start.getTime() || submittedAtGuard.getTime() >= marshalWindow.end.getTime())) {
        logError_(rallyName, 'handleMarshalSubmit_AllDays_', 'FORM_OUTSIDE_DAY_WINDOW', { day: dayVal, dateLabel: tag.dateLabel, submittedAt: submittedAtGuard.toISOString() });
        return;
      }
    }

    var sheetName = 'Marshal Day ' + dayVal;
    var sheet = ss.getSheetByName(sheetName);
    if (!sheet) {
      logError_('handleMarshalSubmit_AllDays_', 'Missing sheet "' + sheetName + '" in rally spreadsheet');
      return;
    }

    var itemResponses = (e && e.response && e.response.getItemResponses)
      ? e.response.getItemResponses()
      : [];
    var byTitle = {};
    for (var i = 0; i < itemResponses.length; i++) {
      var ir = itemResponses[i];
      var it = ir.getItem && ir.getItem();
      var t = it && it.getTitle ? it.getTitle() : '';
      byTitle[t] = ir.getResponse();
    }

    var marshalNumberRaw = (byTitle['Control Number'] || byTitle['Marshal Number'] || '').toString().trim();
    if (!marshalNumberRaw) return;

    // Map by header label (robust against interleaving Marshal/Open columns).
    // Supports numeric "2" => "Marshal 2", and "OC1" => "Open Control 1" if present.
    var headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getDisplayValues();
    headers = (headers && headers[0]) ? headers[0] : [];
    function findColByLabelRegex_(re) {
      for (var h = 0; h < headers.length; h++) {
        var v = (headers[h] || '').toString().trim();
        if (re.test(v)) return h + 1; // 1-based column
      }
      return 0;
    }

    var targetCol = 0;
    var ocMatch = /^\s*OC\s*(\d+)\s*$/i.exec(marshalNumberRaw);
    if (ocMatch) {
      var ocN = parseInt(ocMatch[1], 10);
      if (isFinite(ocN) && ocN > 0) {
        targetCol = findColByLabelRegex_(new RegExp('^\\s*Open\\s*Control\\s*' + ocN + '\\b', 'i'));
      }
    } else {
      var marshalColIndex = parseInt(marshalNumberRaw, 10);
      if (!isFinite(marshalColIndex) || marshalColIndex <= 0) return;
      targetCol = findColByLabelRegex_(new RegExp('^\\s*Marshal\\s*' + marshalColIndex + '\\b', 'i'));
      // Back-compat fallback (older sheets where header labels matched numeric position).
      if (!targetCol) targetCol = 1 + marshalColIndex;
    }

    if (!targetCol) return;

    var carTimes = [];
    var carNo = 1;
    while (true) {
      var key = 'Car ' + carNo + ' Time';
      if (!(key in byTitle)) break;
      var rawVal = (byTitle[key] || '').toString().trim();
      var normalised = parseFormDigitsTimeToHms_(rawVal);
      carTimes.push(normalised != null ? normalised : rawVal);
      carNo++;
    }

    if (carTimes.length === 0) return;

    var numCars = carTimes.length;
    var hashRange = sheet.getRange(2, 1, numCars, 1);
    var hashValues = hashRange.getValues();
    for (var r = 0; r < numCars; r++) {
      var current = hashValues[r][0];
      if (current === '' || current == null) {
        hashValues[r][0] = r + 1;
      }
    }
    hashRange.setValues(hashValues);

    var colRange = sheet.getRange(2, targetCol, numCars, 1);
    var colValues = [];
    for (var r2 = 0; r2 < numCars; r2++) {
      colValues.push([carTimes[r2] || '']);
    }
    colRange.setValues(colValues);
  } catch (err) {
    logError_('handleMarshalSubmit_AllDays_', err);
  }
}

/**
 * Status helper for Controls / forms presence.
 *
 * IMPORTANT:
 * - Read-only: must not create new Drive files/folders or sheets.
 * - Day-scoped: all fields reflect ONLY the requested day.
 *
 * Returns (for selected day n):
 *  - hasControlsData (day-only)
 *  - entrantFormUrl (STC form URL for day)
 *  - marshalFormUrl (Marshal form URL for day)
 *  - responsesFolderId OR responsesFolderUrl (best-effort)
 */
function controlsStatus_(rallyName, day) {
  try {
    var name = rallyName || 'Rally';
    var dayVal = String(day || '1');

    // Spreadsheet must already exist; do not create anything here.
    var ss = findRallySpreadsheet_(name);
    if (!ss) {
      return {
        ok: true,
        rallyName: name,
        day: dayVal,
        spreadsheetUrl: '',
        hasControlsData: false,
        entrantFormUrl: '',
        marshalFormUrl: '',
        responsesFolderId: '',
        responsesFolderUrl: '',
        responsesRootFolderId: '',
        responsesRootFolderUrl: '',
        stcFormResponsesUrl: '',
        stcPwaResponsesUrl: ''
      };
    }

    // Day-scoped controls data: presence of the specific day sheets.
    var stcDaySheet = ss.getSheetByName('STC Day ' + dayVal);
    var marshalDaySheet = ss.getSheetByName('Marshal Day ' + dayVal);
    var hasControlsData = !!(stcDaySheet || marshalDaySheet);
    var controlsSpreadsheetUrl = stcDaySheet ? sheetUrl_(ss, stcDaySheet) : ss.getUrl();

    // Best-effort: find the rally folder without creating duplicates.
    var rallyFolder = findRallyFolder_(name);

    // Day-scoped form URLs (read-only).
    var urls = resolveDayFormUrls_(rallyFolder, name, dayVal);

    // Best-effort: response folder for this day (prefer response sheets' parent folder).
    var respFolder = resolveResponsesFolderForDay_(rallyFolder, name, dayVal);

    // Non-day-based: parent folder above the per-day responses folder (best-effort).
    // This is the folder you can open to see all response forms/sheets for the rally.
    var respRoot = { responsesRootFolderId: '', responsesRootFolderUrl: '' };
    try {
      if (respFolder && respFolder.responsesFolderId) {
        var f = DriveApp.getFolderById(respFolder.responsesFolderId);
        respRoot.responsesRootFolderId = f.getId();
          respRoot.responsesRootFolderUrl = f.getUrl();
      }
    } catch (eRoot) {}
    try {
      if (!respRoot.responsesRootFolderId && rallyFolder) {
        respRoot.responsesRootFolderId = rallyFolder.getId();
        respRoot.responsesRootFolderUrl = rallyFolder.getUrl();
      }
    } catch (eRoot2) {}


    // Fallback: if we couldn't get a parent, use the rally folder itself.
    try {
      if ((!respRoot.responsesRootFolderId || !respRoot.responsesRootFolderUrl) && rallyFolder) {
        respRoot.responsesRootFolderId = rallyFolder.getId();
        respRoot.responsesRootFolderUrl = rallyFolder.getUrl();
      }
    } catch (eRoot2) {}


    return {
      ok: true,
      rallyName: name,
      day: dayVal,
      spreadsheetUrl: controlsSpreadsheetUrl,
      hasControlsData: hasControlsData,
      entrantFormUrl: urls.entrantFormUrl || '',
      marshalFormUrl: urls.marshalFormUrl || '',
      responsesFolderId: respFolder.responsesFolderId || '',
      responsesFolderUrl: respFolder.responsesFolderUrl || '',
      responsesRootFolderId: respRoot.responsesRootFolderId || '',
      responsesRootFolderUrl: respRoot.responsesRootFolderUrl || '',
      stcFormResponsesUrl: getStcFormResponsesUrl_(rallyFolder, name, dayVal),
      stcPwaResponsesUrl: getStcPwaResponsesUrl_(rallyFolder, name, dayVal)
    };
  } catch (e) {
    return { ok: false, error: String(e) };
  }
}

/**

 * Does NOT create folders.

  var props = getProps_();
  var rallyFolderIdKey = PROP_PREFIX + 'rallyFolderId_' + slug;
  var rallyFolderId = props.getProperty(rallyFolderIdKey);

  // Fast path: use the persisted rally-folder ID.
  if (rallyFolderId) {
    try { return DriveApp.getFolderById(rallyFolderId); } catch (e) {}
  }

  // Compatibility fallback only: search the existing base folder without creating or locking.
  var baseFolder = findBaseFolderReadOnly_();
  if (!baseFolder) return null;

    } catch (e2) {}
  }

  return null;
}

/**
 * Resolve per-day STC/Marshal form URLs without creating forms.
 * Uses the same naming convention as initForms_ / buildFormsForControls_.
 */
function resolveDayFormUrls_(rallyFolder, rallyName, dayVal) {
  var out = { entrantFormUrl: '', marshalFormUrl: '' };
  if (!rallyFolder) return out;

  var stcDriveNameBase = 'STC – Day ' + dayVal + ' – ' + rallyName;
  var marshalDriveNameBase = 'Marshals – Day ' + dayVal + ' – ' + rallyName;
  var stcNames = nameVariants_(stcDriveNameBase);
  var marshalNames = nameVariants_(marshalDriveNameBase);

  try {
    var stcFile = firstFileByNameVariants_(rallyFolder, stcNames);
    if (stcFile) {
      out.entrantFormUrl = FormApp.openById(stcFile.getId()).getPublishedUrl();
    }

    var marshalFile = firstFileByNameVariants_(rallyFolder, marshalNames);
    if (marshalFile) {
      out.marshalFormUrl = FormApp.openById(marshalFile.getId()).getPublishedUrl();
    }
  } catch (eM) {}

}

/**
 * Resolve a folder to open for "View Response forms" for a given day.
 *
 * Best-effort rules:
 *  1) Prefer the parent folder of the STC/Mashal response spreadsheets for that day (if found).
 *  2) Otherwise, fall back to the parent folder of the STC form for that day (if found).
 *  3) Otherwise, fall back to the rally folder itself (if known).
 *
 * Returns both id and url; callers can use either.
 */
function resolveResponsesFolderForDay_(rallyFolder, rallyName, dayVal) {
  var out = { responsesFolderId: '', responsesFolderUrl: '' };

  function setFolder_(folder) {
    if (!folder) return;
    out.responsesFolderId = folder.getId();
    out.responsesFolderUrl = 'https://drive.google.com/drive/folders/' + folder.getId();
  }

  if (!rallyFolder) return out;

  // 1) Response spreadsheets (created by ensureFormHasResponseSheet_)
  var respNameBases = [
    'STC Responses – Day ' + dayVal + ' – ' + rallyName,
    'Marshal Responses – Day ' + dayVal + ' – ' + rallyName
  ];

  for (var i = 0; i < respNameBases.length; i++) {
    try {
      var f = firstFileByNameVariants_(rallyFolder, nameVariants_(respNameBases[i]));
      if (f) {
        var parents = f.getParents();
        if (parents.hasNext()) {
          setFolder_(parents.next());
          return out;
        }
      }
    } catch (e1) {}
  }

  // 2) Parent folder of the STC form (if present)
  try {
    var stcDriveNameBase = 'STC – Day ' + dayVal + ' – ' + rallyName;
    var stcFile = firstFileByNameVariants_(rallyFolder, nameVariants_(stcDriveNameBase));
    if (stcFile) {
      var stcParents = stcFile.getParents();
      if (stcParents.hasNext()) {
        setFolder_(stcParents.next());
        return out;
      }
    }
  } catch (e2) {}

  // 3) Rally folder fallback
  setFolder_(rallyFolder);
  return out;
}

function rallyTrackingHandleRequest_(e) {
  try {
    const p = (e && e.parameter) ? e.parameter : {};
    // v0.24: streamlined live/bulk senders POST raw JSON. Apps Script does not
    // populate e.parameter.mode for that body, so read the payload before routing.
    // GET/query traffic remains unchanged because rallyTrackingReadPayload_ falls
    // back to e.parameter when there is no JSON body.
    const requestPayload = rallyTrackingReadPayload_(e);
    const mode = String(requestPayload.mode || p.mode || 'ping');

    // v3.22: normal Tracking activity self-maintains the 7-day retention window.
    // Confirmation-only calls are excluded so acknowledgements remain lightweight.
    if ([
      'trackingEventUpload',
      'trackingEventRead',
      'trackingStatus',
      'trackingSubmit',
      'liveSubmit',
      'bulkSubmit',
      'trackingRead'
    ].indexOf(mode) !== -1) {
      rallyTrackingEnsureHousekeepingCurrent_();
    }

    if (mode === 'ping') {
      return rallyTrackingJson_({
        ok: true,
        mode: 'ping',
        service: 'Rally Tracking Event Tracking',
        version: RALLY_TRACKING_TRACKING_VERSION
      });
    }

    if (mode === 'trackingSetup') {
      rallyTrackingRequireKey_(requestPayload.key || p.key);
      return rallyTrackingJson_(Object.assign({ ok: true, mode: mode }, rallyTrackingTrackingSetup_()));
    }

    if (mode === 'trackingEventUpload') {
      const payload = requestPayload;
      rallyTrackingRequireKey_(payload.key || p.key);
      return rallyTrackingJson_(Object.assign({ ok: true, mode: mode }, rallyTrackingTrackingEventUpload_(payload)));
    }

    if (mode === 'trackingEventRead') {
      rallyTrackingRequireKey_(p.key);
      return rallyTrackingJson_(Object.assign({ ok: true, mode: mode }, rallyTrackingTrackingEventRead_(p)));
    }

    if (mode === 'trackingStatus') {
      rallyTrackingRequireKey_(p.key);
      return rallyTrackingJson_(Object.assign({ ok: true, mode: mode }, rallyTrackingTrackingStatus_(p)));
    }

    if (mode === 'trackingSubmit' || mode === 'liveSubmit') {
      const payload = requestPayload;
      rallyTrackingRequireKey_(payload.key || p.key);
      const trackPayload = payload.track && typeof payload.track === 'object' ? payload.track : payload;
      return rallyTrackingJson_(Object.assign({ ok: true, mode: mode }, rallyTrackingTrackingSubmit_(trackPayload)));
    }

    if (mode === 'trackConfirm') {
      rallyTrackingRequireKey_(p.key);
      return rallyTrackingJson_(Object.assign({ ok: true, mode: mode }, rallyTrackingTrackConfirm_(p)));
    }

    if (mode === 'bulkSubmit') {
      const payload = requestPayload;
      rallyTrackingRequireKey_(payload.key || p.key);
      return rallyTrackingJson_(Object.assign({ ok: true, mode: mode }, rallyTrackingBulkSubmit_(payload)));
    }

    if (mode === 'bulkConfirm') {
      rallyTrackingRequireKey_(p.key);
      return rallyTrackingJson_(Object.assign({ ok: true, mode: mode }, rallyTrackingBulkConfirm_(p)));
    }

    if (mode === 'trackingConfirm') {
      rallyTrackingRequireKey_(p.key);
      return rallyTrackingJson_(Object.assign({ ok: true, mode: mode }, rallyTrackingTrackingConfirm_(p)));
    }

    if (mode === 'trackingRead') {
      rallyTrackingRequireKey_(p.key);
      return rallyTrackingJson_(Object.assign({ ok: true, mode: mode }, rallyTrackingTrackingRead_(p)));
    }

    if (mode === 'trackingHousekeeping') {
      rallyTrackingRequireKey_(p.key);
      return rallyTrackingJson_(Object.assign({ ok: true, mode: mode }, rallyTrackingTrackingHousekeeping_()));
    }

    return rallyTrackingJson_({ ok: false, error: 'Unknown mode: ' + mode });
  } catch (err) {
    rallyTrackingLogError_('rallyTrackingHandleRequest_', err, { parameters: e && e.parameter ? e.parameter : {} });
    return rallyTrackingJson_({ ok: false, error: String(err && err.message ? err.message : err) });
  }
}

function rallyTrackingReadPayload_(e) {
  const p = (e && e.parameter) ? e.parameter : {};
  let raw = p.data || '';

  if (!raw && e && e.postData && e.postData.contents) {
    raw = e.postData.contents;
  }

  if (!raw) return Object.assign({}, p);

  try {
    const parsed = JSON.parse(raw);
    return Object.assign({}, p, parsed || {});
  } catch (err) {
    throw new Error('Invalid tracking payload JSON');
  }
}

  if (String(value || '') !== RALLY_TRACKING_KEY) {
    throw new Error('Invalid Rally Tracking test key');
  }
}

function rallyTrackingTrackingEventUpload_(payload) {
  const eventId = String(payload.eventId || '').trim();
  if (!eventId) throw new Error('eventId is required');
  const entrants = Array.isArray(payload.entrants) ? payload.entrants : [];
  const routeData = payload.routeData && typeof payload.routeData === 'object' ? payload.routeData : {};
  const uploadedAt = new Date();
  const rallyTitle = String(payload.rallyTitle || '').trim();
  if (!rallyTitle) throw new Error('rallyTitle is required');
  rallyTrackingRememberEventRally_(eventId, rallyTitle);
  const storage = rallyTrackingEnsureStorage_(rallyTitle, eventId);
  const sheet = storage.eventsSheet;
  const lock = LockService.getScriptLock();
  lock.waitLock(30000);
  try {
    const lastRow = sheet.getLastRow();
    let targetRow = lastRow + 1;
    if (lastRow > 1) {
      const ids = sheet.getRange(2, 1, lastRow - 1, 1).getValues();
      for (let i = 0; i < ids.length; i++) {
        if (String(ids[i][0] || '') === eventId) { targetRow = i + 2; break; }
      }
    }
    sheet.getRange(targetRow, 1, 1, RALLY_TRACKING_EVENT_HEADERS.length).setValues([[
      eventId,
      uploadedAt,
      String(payload.pageHeader || '').trim(),
      String(payload.rallyTitle || '').trim(),
      String(payload.day || '').trim(),
      String(payload.route || '').trim(),
      entrants.length,
      JSON.stringify(entrants),
      JSON.stringify(routeData)
    ]]);
    return { eventId: eventId, uploadedAt: uploadedAt.toISOString(), entrantCount: entrants.length, replaced: targetRow <= lastRow };
  } finally {
    try { lock.releaseLock(); } catch (_) {}
  }
}

function rallyTrackingTrackingSubmit_(payload) {
  const serverEntryMs = Date.now();
  const clean = rallyTrackingValidateTrackingPayload_(payload);
  const validatedMs = Date.now();

  // v0.22 FAST RECEIPT CONTRACT:
  // Do not inspect Tracking history before accepting a live report.
  // The phone owns the original report identity and may resend the exact same packet
  // when an acknowledgement is uncertain. Any duplicate raw receipts are reconciled
  // after receipt by the one-minute worker and filtered by trackingRead immediately.
  rallyTrackingRememberEventRally_(clean.eventId, clean.rallyTitle);
  const trackingSheet = rallyTrackingOpenTrackingSheet_(clean.rallyTitle, clean.eventId);
  const sheetReadyMs = Date.now();
  const gpsTime = new Date(clean.gpsTime);
  const reportKey = rallyTrackingReportKey_(clean);
  const packetId = rallyTrackingPacketId_(payload, reportKey);
  const received = new Date();

  trackingSheet.appendRow([
    received,
    gpsTime,
    clean.eventId,
    clean.rallyTitle,
    clean.day,
    clean.route,
    clean.rallyNo,
    clean.driverName,
    clean.latitude,
    clean.longitude,
    clean.accuracy,
    clean.speedKph,
    clean.senderId,
    clean.sequence,
    reportKey
  ]);
  const appendCompletedMs = Date.now();

  // Best-effort recent receipt marker. Confirmation correctness does not depend on
  // this cache: a cache miss falls back to the exact Tracking-sheet lookup.
  let receiptCacheWritten = false;
  const cacheStartedMs = Date.now();
  try {
    if (reportKey) {
      CacheService.getScriptCache().put(
        rallyTrackingLiveReceiptCacheKey_(reportKey),
        '1',
        RALLY_TRACKING_LIVE_RECEIPT_CACHE_SECONDS
      );
      receiptCacheWritten = true;
    }
  } catch (_) {}
  // Streamlined sender receipt: Packet ID is phone-facing identity; Report Key remains
  // the authoritative Tracking-sheet identity used by the Rally Tracking service.
  try { rallyTrackingSaveTrackReceipt_(packetId, reportKey); } catch (_) {}
  const responsePreparedMs = Date.now();

  return {
    accepted: true,
    duplicate: false,
    stored: true,
    reportKey: reportKey,
    packetId: packetId,
    rowNumber: null,
    eventId: clean.eventId,
    day: clean.day,
    route: clean.route,
    rallyNo: clean.rallyNo,
    gpsTime: gpsTime.toISOString(),
    receivedTime: received.toISOString(),
    appendCompletedTime: new Date(appendCompletedMs).toISOString(),
    receiptCacheWritten: receiptCacheWritten,
    serverTiming: {
      entryTime: new Date(serverEntryMs).toISOString(),
      validateMs: Math.max(0, validatedMs - serverEntryMs),
      sheetOpenMs: Math.max(0, sheetReadyMs - validatedMs),
      appendMs: Math.max(0, appendCompletedMs - sheetReadyMs),
      receiptCacheMs: Math.max(0, responsePreparedMs - cacheStartedMs),
      totalCodeMs: Math.max(0, responsePreparedMs - serverEntryMs)
    }
  };
}

function rallyTrackingLiveReceiptCacheKey_(reportKey) {
  const digest = Utilities.computeDigest(
    Utilities.DigestAlgorithm.SHA_256,
    String(reportKey || ''),
    Utilities.Charset.UTF_8
  );
  const hex = digest.map(function(byte) {
    const n = byte < 0 ? byte + 256 : byte;
    return ('0' + n.toString(16)).slice(-2);
  }).join('');
  return RALLY_TRACKING_LIVE_RECEIPT_CACHE_PREFIX + hex;
}

function rallyTrackingRecentReceiptExists_(reportKey) {
  if (!reportKey) return false;
  try {
    return CacheService.getScriptCache().get(
      rallyTrackingLiveReceiptCacheKey_(reportKey)
    ) === '1';
  } catch (_) {
    return false;
  }
}

function rallyTrackingReportKey_(clean) {
  if (!clean.senderId || clean.sequence === '' || !clean.gpsTime) return '';
  const gpsTime = new Date(clean.gpsTime);
  if (isNaN(gpsTime.getTime())) return '';
  return [clean.eventId, clean.senderId, String(clean.sequence), gpsTime.toISOString()].join('|');
}

function rallyTrackingReportKeyExists_(sheet, reportKey) {
  if (!reportKey) return false;
  const lastRow = sheet.getLastRow();
  if (lastRow <= 1) return false;
  const keyColumn = RALLY_TRACKING_TRACKING_HEADERS.indexOf('Report Key') + 1;
  if (keyColumn <= 0) return false;

  const match = sheet
    .getRange(2, keyColumn, lastRow - 1, 1)
    .createTextFinder(reportKey)
    .matchEntireCell(true)
    .findNext();

  return !!match;
}

function rallyTrackingTrackingConfirm_(p) {
  const serverEntryMs = Date.now();
  const reportKey = String((p && p.reportKey) || '').trim();
  if (!reportKey) throw new Error('reportKey is required');

  const cacheCheckedMs = Date.now();
  if (rallyTrackingRecentReceiptExists_(reportKey)) {
    const completedMs = Date.now();
    return {
      confirmed: true,
      reportKey: reportKey,
      source: 'recent-receipt-cache',
      serverTiming: {
        entryTime: new Date(serverEntryMs).toISOString(),
        cacheMs: Math.max(0, completedMs - cacheCheckedMs),
        sheetOpenMs: 0,
        lookupMs: 0,
        totalCodeMs: Math.max(0, completedMs - serverEntryMs)
      }
    };
  }

  // Compatibility/safety fallback for an older receipt, cache eviction or script restart.
  const confirmEventId = String(reportKey.split('|')[0] || '').trim();
  const trackingSheet = rallyTrackingOpenTrackingSheet_('', confirmEventId);
  const sheetReadyMs = Date.now();
  const confirmed = rallyTrackingReportKeyExists_(trackingSheet, reportKey);
  const lookupCompletedMs = Date.now();

  return {
    confirmed: confirmed,
    reportKey: reportKey,
    source: 'sheet-fallback',
    serverTiming: {
      entryTime: new Date(serverEntryMs).toISOString(),
      cacheMs: Math.max(0, cacheCheckedMs - serverEntryMs),
      sheetOpenMs: Math.max(0, sheetReadyMs - cacheCheckedMs),
      lookupMs: Math.max(0, lookupCompletedMs - sheetReadyMs),
      totalCodeMs: Math.max(0, lookupCompletedMs - serverEntryMs)
    }
  };
}

function rallyTrackingValidateTrackingPayload_(p) {
  const latitude = Number(p.latitude);
  const longitude = Number(p.longitude);
  const gpsTime = String(p.gpsTime || '');

  if (!String(p.eventId || '').trim()) throw new Error('eventId is required');
  if (!String(p.day || '').trim()) throw new Error('day is required');
  if (!String(p.route || '').trim()) throw new Error('route is required');
  if (!String(p.rallyNo || '').trim()) throw new Error('rallyNo is required');
  if (!Number.isFinite(latitude) || latitude < -90 || latitude > 90) throw new Error('Invalid latitude');
  if (!Number.isFinite(longitude) || longitude < -180 || longitude > 180) throw new Error('Invalid longitude');
  if (!gpsTime || isNaN(new Date(gpsTime).getTime())) throw new Error('Invalid gpsTime');

  return {
    eventId: String(p.eventId).trim(),
    rallyTitle: String(p.rallyTitle || '').trim(),
    day: String(p.day).trim(),
    route: String(p.route).trim(),
    rallyNo: String(p.rallyNo).trim(),
    driverName: String(p.driverName || '').trim(),
    latitude: latitude,
    longitude: longitude,
    accuracy: rallyTrackingOptionalNumber_(p.accuracy),
    speedKph: rallyTrackingOptionalNumber_(p.speedKph !== undefined ? p.speedKph : p.speed),
    senderId: String(p.senderId || '').trim(),
    sequence: rallyTrackingOptionalNumber_(p.sequence),
    gpsTime: gpsTime
  };
}

function rallyTrackingOptionalNumber_(value) {
  if (value === '' || value === null || value === undefined) return '';
  const n = Number(value);
  return Number.isFinite(n) ? n : '';
}

function rallyTrackingTrackingRead_(p) {
  const eventId = String(p.eventId || '').trim();
  const storage = rallyTrackingEnsureStorage_('', eventId);
  const eventAliases = rallyTrackingEventIdAliases_(storage, eventId);
  const values = storage.trackingSheet.getDataRange().getValues();
  if (values.length <= 1) {
    return { points: [], count: 0, generatedAt: new Date().toISOString() };
  }

  const day = String(p.day || '').trim();
  const route = String(p.route || '').trim();
  const rallyNo = String(p.rallyNo || '').trim();
  const latestOnly = String(p.latestOnly || '').toLowerCase() === 'true';
  const limit = Math.max(1, Math.min(50000, Number(p.limit) || 2000));
  const trackingWindow = rallyTrackingTrackingDayWindow_(p.trackingDate || p.eventDate);

  const seenReportKeys = {};
  let points = values.slice(1).filter(function (rawRow) {
    const reportKey = String(rawRow[14] || '');
    if (!reportKey) return true;
    if (seenReportKeys[reportKey]) return false;
    seenReportKeys[reportKey] = true;
    return true;
  }).map(rallyTrackingTrackingRowToObject_).filter(function (row) {
    const gpsMs = new Date(row.gpsTime).getTime();
    if (!Number.isFinite(gpsMs)) return false;
    if (gpsMs < trackingWindow.start.getTime() || gpsMs >= trackingWindow.end.getTime()) return false;
    if (eventId && !eventAliases[String(row.eventId || '').trim()]) return false;
    if (day && row.day !== day) return false;
    if (route && row.route !== route) return false;
    if (rallyNo && row.rallyNo !== rallyNo) return false;
    return true;
  });

  points.sort(function (a, b) {
    return new Date(a.gpsTime).getTime() - new Date(b.gpsTime).getTime();
  });

  if (latestOnly) {
    const latest = {};
    points.forEach(function (row) {
      const key = row.eventId + '|' + row.day + '|' + row.route + '|' + row.rallyNo;
      latest[key] = row;
    });
    points = Object.keys(latest).map(function (key) { return latest[key]; });
  }

  if (points.length > limit) points = points.slice(points.length - limit);

  return {
    version: RALLY_TRACKING_TRACKING_VERSION,
    generatedAt: new Date().toISOString(),
    trackingDay: trackingWindow.label,
    trackingWindowStart: trackingWindow.start.toISOString(),
    trackingWindowEnd: trackingWindow.end.toISOString(),
    trackingTimeZone: trackingWindow.timeZone,
    count: points.length,
    points: points
  };
}

function rallyTrackingTrackingHousekeeping_() {
  const startedMs = Date.now();
  const props = PropertiesService.getScriptProperties();
  let rallyTitles = [];
  try { rallyTitles = JSON.parse(props.getProperty(RALLY_TRACKING_PROP_PREFIX + 'projectRallies') || '[]'); } catch (_) { rallyTitles = []; }
  if (!Array.isArray(rallyTitles)) rallyTitles = [];

  const currentWindow = rallyTrackingTrackingDayWindowForDate_(new Date());
  let beforeTrackingRows = 0;
  let afterTrackingRows = 0;
  let deletedTrackingRows = 0;
  let duplicatesRemoved = 0;
  let projectsChecked = 0;

  const lock = LockService.getScriptLock();
  lock.waitLock(30000);
  try {
    rallyTitles.forEach(function(title) {
      try {
        const storage = rallyTrackingEnsureStorage_(title, '');
        projectsChecked++;
        const sheet = storage.trackingSheet;
        beforeTrackingRows += Math.max(0, sheet.getLastRow() - 1);
        const liveDedupe = rallyTrackingDeduplicateTracking_(sheet);
        duplicatesRemoved += Number(liveDedupe.duplicatesRemoved || 0);
        deletedTrackingRows += rallyTrackingDeleteTrackingRowsBefore_(sheet, currentWindow);
        afterTrackingRows += Math.max(0, sheet.getLastRow() - 1);
      } catch (_) {}
    });

    const receiptCleanup = rallyTrackingCleanupStreamlinedReceipts_();
    props.setProperty(RALLY_TRACKING_LAST_CLEANUP_PROP, currentWindow.label);

    rallyTrackingRemoveObsoleteReplayTrigger_();
    const cleanupTrigger = rallyTrackingEnsureDailyCleanupTrigger_();

    return {
      trackingDay: currentWindow.label,
      projectsChecked: projectsChecked,
      liveDuplicatesRemoved: duplicatesRemoved,
      beforeTrackingRows: beforeTrackingRows,
      trackingRowsRemoved: deletedTrackingRows,
      afterTrackingRows: afterTrackingRows,
      trackReceiptsRemoved: receiptCleanup.trackRemoved,
      bulkReceiptsRemoved: receiptCleanup.bulkRemoved,
      cleanupTrigger: cleanupTrigger,
      totalCodeMs: Math.max(0, Date.now() - startedMs)
    };
  } finally {
    try { lock.releaseLock(); } catch (_) {}
  }
}

function rallyTrackingRunHousekeeping() {
  return rallyTrackingTrackingHousekeeping_();
}

function rallyTrackingDeleteTrackingRowsBefore_(sheet, currentWindow) {
  const lastRow = sheet.getLastRow();
  if (lastRow <= 1) return 0;

  const keepDays = Math.max(1, Number(RALLY_TRACKING_KEEP_TRACKING_DAYS) || 7);
  const cutoffMs = currentWindow.start.getTime() - (keepDays - 1) * 24 * 60 * 60 * 1000;
  const gpsValues = sheet.getRange(2, 2, lastRow - 1, 1).getValues();
  const rowsToDelete = [];

  gpsValues.forEach(function (row, index) {
    const gpsMs = new Date(row[0]).getTime();
    if (Number.isFinite(gpsMs) && gpsMs < cutoffMs) rowsToDelete.push(index + 2);
  });

  if (!rowsToDelete.length) return 0;

  // Delete contiguous row groups from the bottom so row numbers remain valid.
  let groupEnd = rowsToDelete[rowsToDelete.length - 1];
  let groupStart = groupEnd;

  for (let i = rowsToDelete.length - 2; i >= -1; i--) {
    const rowNo = i >= 0 ? rowsToDelete[i] : null;
    if (rowNo !== null && rowNo === groupStart - 1) {
      groupStart = rowNo;
      continue;
    }

    sheet.deleteRows(groupStart, groupEnd - groupStart + 1);
    if (rowNo !== null) {
      groupStart = rowNo;
      groupEnd = rowNo;
    }
  }

  return rowsToDelete.length;
}

function rallyTrackingDigestKey_(prefix, value) {
  const digest = Utilities.computeDigest(
    Utilities.DigestAlgorithm.SHA_256,
    String(value || ''),
    Utilities.Charset.UTF_8
  );
  const hex = digest.map(function(byte) {
    const n = byte < 0 ? byte + 256 : byte;
    return ('0' + n.toString(16)).slice(-2);
  }).join('');
  return String(prefix || '') + hex;
}

function rallyTrackingSaveBulkReceipt_(packageId, recordCount, appended, duplicateCount) {
  const receipt = {
    packageId: String(packageId),
    recordCount: Number(recordCount || 0),
    appended: Number(appended || 0),
    duplicates: Number(duplicateCount || 0),
    receivedAt: new Date().toISOString()
  };
  const raw = JSON.stringify(receipt);

  // v0.25: confirmation is time-critical. Put the compact package receipt in
  // Script Cache first so a full/busy properties store cannot hide a successful
  // Tracking-sheet commit from the sender. Six hours comfortably covers field retry.
  let cacheWritten = false;
  try {
    CacheService.getScriptCache().put(
      rallyTrackingBulkReceiptCacheKey_(packageId),
      raw,
      21600
    );
    cacheWritten = true;
  } catch (_) {}

  // Keep the established 24-hour Script Properties receipt as a secondary path.
  // It is deliberately best-effort: the cache receipt is sufficient for the
  // immediate positive acknowledgement that lets the phone clear its package.
  let propertyWritten = false;
  try {
    PropertiesService.getScriptProperties().setProperty(
      rallyTrackingBulkReceiptPropertyKey_(packageId),
      raw
    );
    propertyWritten = true;
  } catch (_) {}

  if (!cacheWritten && !propertyWritten) {
    throw new Error('Bulk committed but package receipt could not be recorded');
  }
  return { cacheWritten: cacheWritten, propertyWritten: propertyWritten };
}

function rallyTrackingBulkConfirm_(p) {
  const packageId = String((p && p.packageId) || '').trim();
  if (!packageId) throw new Error('packageId is required');

  let raw = '';
  let source = '';
  try {
    raw = CacheService.getScriptCache().get(rallyTrackingBulkReceiptCacheKey_(packageId)) || '';
    if (raw) source = 'bulk-receipt-cache';
  } catch (_) {}

  if (!raw) {
    try {
      raw = PropertiesService.getScriptProperties().getProperty(
        rallyTrackingBulkReceiptPropertyKey_(packageId)
      ) || '';
      if (raw) source = 'bulk-receipt-property';
    } catch (_) {}
  }

  if (!raw) return { confirmed: false, packageId: packageId };
  try {
    const receipt = JSON.parse(raw);
    return {
      confirmed: !!(receipt && String(receipt.packageId || '') === packageId),
      packageId: packageId,
      recordCount: Number(receipt && receipt.recordCount || 0),
      appended: Number(receipt && receipt.appended || 0),
      duplicates: Number(receipt && receipt.duplicates || 0),
      receivedAt: receipt && receipt.receivedAt ? String(receipt.receivedAt) : '',
      source: source
    };
  } catch (_) {
    return { confirmed: false, packageId: packageId };
  }
}

function rallyTrackingBulkSubmit_(payload) {
  payload = payload || {};
  const packageId = String(payload.packageId || '').trim();
  if (!packageId) throw new Error('packageId is required');
  const tracks = Array.isArray(payload.tracks) ? payload.tracks : [];
  if (!tracks.length) throw new Error('bulk tracks missing');
  if (tracks.length > 1000) throw new Error('bulk too large');

  const firstTrack = tracks[0] || {};
  const bulkRallyTitle = String(firstTrack.rallyTitle || '').trim();
  const bulkEventId = String(firstTrack.eventId || '').trim();
  if (!bulkRallyTitle) throw new Error('bulk rallyTitle is required');
  rallyTrackingRememberEventRally_(bulkEventId, bulkRallyTitle);
  const storage = rallyTrackingEnsureStorage_(bulkRallyTitle, bulkEventId);
  const sheet = storage.trackingSheet;
  const cleanTracks = tracks.map(function(raw) {
    const clean = rallyTrackingValidateTrackingPayload_(raw);
    const reportKey = rallyTrackingReportKey_(clean);
    if (!reportKey) throw new Error('bulk track Report Key missing');
    return { raw: raw, clean: clean, reportKey: reportKey };
  });

  // Event End is not latency critical. Read the Report Key column once, then append
  // only records that have not already arrived through the live path.
  const lastRow = sheet.getLastRow();
  const existing = {};
  if (lastRow > 1) {
    const keyColumn = RALLY_TRACKING_TRACKING_HEADERS.indexOf('Report Key') + 1;
    sheet.getRange(2, keyColumn, lastRow - 1, 1).getDisplayValues().forEach(function(row) {
      const key = String(row[0] || '');
      if (key) existing[key] = true;
    });
  }

  const received = new Date();
  const rows = [];
  let duplicates = 0;
  cleanTracks.forEach(function(item) {
    const clean = item.clean;
    const reportKey = item.reportKey;
    if (existing[reportKey]) {
      duplicates++;
    } else {
      rows.push([
        received,
        new Date(clean.gpsTime),
        clean.eventId,
        clean.rallyTitle,
        clean.day,
        clean.route,
        clean.rallyNo,
        clean.driverName,
        clean.latitude,
        clean.longitude,
        clean.accuracy,
        clean.speedKph,
        clean.senderId,
        clean.sequence,
        reportKey
      ]);
      existing[reportKey] = true;
    }

    // v0.25: Event End bulk owns one package receipt. Do not create hundreds of
    // individual persistent receipts/cache entries here. Live submissions retain
    // their existing individual acknowledgement path unchanged.
  });

  if (rows.length) {
    const startRow = sheet.getLastRow() + 1;
    sheet.getRange(startRow, 1, rows.length, RALLY_TRACKING_TRACKING_HEADERS.length).setValues(rows);
    sheet.getRange(startRow, 1, rows.length, 2).setNumberFormat('dd/MM/yyyy HH:mm:ss');
  }

  rallyTrackingSaveBulkReceipt_(packageId, tracks.length, rows.length, duplicates);

  return {
    accepted: true,
    stored: true,
    packageId: packageId,
    recordCount: tracks.length,
    appended: rows.length,
    duplicates: duplicates,
    trackingRows: Math.max(0, sheet.getLastRow() - 1),
    receivedTime: received.toISOString(),
    version: RALLY_TRACKING_TRACKING_VERSION
  };
}

function rallyTrackingDeduplicateTracking_(sheet) {
  const lastRow = sheet.getLastRow();
  if (lastRow <= 2) return { duplicatesRemoved: 0 };

  const keyColumn = RALLY_TRACKING_TRACKING_HEADERS.indexOf('Report Key') + 1;
  if (keyColumn <= 0) return { duplicatesRemoved: 0 };

  const values = sheet.getRange(2, keyColumn, lastRow - 1, 1).getValues();
  const seen = {};
  const duplicateRows = [];

  values.forEach(function(row, index) {
    const key = String(row[0] || '');
    if (!key) return;
    if (seen[key]) {
      duplicateRows.push(index + 2);
      return;
    }
    seen[key] = true;
  });

  rallyTrackingDeleteRowsInBlocks_(sheet, duplicateRows);
  return { duplicatesRemoved: duplicateRows.length };
}

function rallyTrackingDeleteRowsInBlocks_(sheet, rowNumbers) {
  if (!rowNumbers || !rowNumbers.length) return;

  rowNumbers.sort(function(a, b) { return b - a; });
  let blockHigh = rowNumbers[0];
  let blockLow = blockHigh;

  for (let i = 1; i < rowNumbers.length; i++) {
    const row = rowNumbers[i];
    if (row === blockLow - 1) {
      blockLow = row;
      continue;
    }
    sheet.deleteRows(blockLow, blockHigh - blockLow + 1);
    blockHigh = row;
    blockLow = row;
  }
  sheet.deleteRows(blockLow, blockHigh - blockLow + 1);
}

function rallyTrackingCleanupStreamlinedReceipts_() {
  const props = PropertiesService.getScriptProperties();
  const all = props.getProperties();
  const cutoff = Date.now() - Math.max(1, Number(RALLY_TRACKING_STREAMLINED_RECEIPT_HOURS) || 24) * 60 * 60 * 1000;
  let trackRemoved = 0;
  let bulkRemoved = 0;

  Object.keys(all).forEach(function(key) {
    const isTrack = key.indexOf(RALLY_TRACKING_PACKET_RECEIPT_PREFIX) === 0;
    const isBulk = key.indexOf(RALLY_TRACKING_BULK_RECEIPT_PREFIX) === 0;
    if (!isTrack && !isBulk) return;
    let when = 0;
    try {
      const obj = JSON.parse(all[key]);
      when = new Date(obj.receivedAt || obj.updatedAt || '').getTime();
    } catch (_) {}
    if (when && when >= cutoff) return;
    try {
      props.deleteProperty(key);
      if (isTrack) trackRemoved++; else bulkRemoved++;
    } catch (_) {}
  });
  return { trackRemoved: trackRemoved, bulkRemoved: bulkRemoved };
}

function rallyTrackingTrackingDayWindowForDate_(dateValue) {
  const timeZone = RALLY_TRACKING_TRACKING_TIME_ZONE || Session.getScriptTimeZone() || 'GMT';
  const date = (dateValue instanceof Date) ? dateValue : new Date(dateValue);
  const shifted = new Date(date.getTime() - RALLY_TRACKING_TRACKING_DAY_START_HOUR * 60 * 60 * 1000);
  const label = Utilities.formatDate(shifted, timeZone, 'yyyy-MM-dd');
  return rallyTrackingTrackingDayWindow_(label);
}

function rallyTrackingTrackingDayWindow_(requestedDate) {
  const timeZone = RALLY_TRACKING_TRACKING_TIME_ZONE || Session.getScriptTimeZone() || 'GMT';
  let label = String(requestedDate || '').trim();

  if (!/^\d{4}-\d{2}-\d{2}$/.test(label)) {
    const shiftedNow = new Date(Date.now() - RALLY_TRACKING_TRACKING_DAY_START_HOUR * 60 * 60 * 1000);
    label = Utilities.formatDate(shiftedNow, timeZone, 'yyyy-MM-dd');
  }

  const hh = ('0' + RALLY_TRACKING_TRACKING_DAY_START_HOUR).slice(-2);
  const start = Utilities.parseDate(label + ' ' + hh + ':00:00', timeZone, 'yyyy-MM-dd HH:mm:ss');
  const end = new Date(start.getTime() + 24 * 60 * 60 * 1000);

  return { label: label, start: start, end: end, timeZone: timeZone };
}

function rallyTrackingEnsureStorage_(rallyTitle, eventId) {
  let title = String(rallyTitle || '').trim();
  if (!title) title = rallyTrackingRallyForEvent_(eventId);
  if (!title) {
    throw new Error('Tracking rally project could not be resolved.');
  }

  const project = ensureRallyStorage_(title);
  rallyTrackingRememberRally_(title);

  let eventsSheet = project.spreadsheet.getSheetByName(RALLY_TRACKING_EVENTS_SHEET);
  if (!eventsSheet) eventsSheet = project.spreadsheet.insertSheet(RALLY_TRACKING_EVENTS_SHEET);
  rallyTrackingEnsureHeaders_(eventsSheet, RALLY_TRACKING_EVENT_HEADERS);

  let trackingSheet = project.spreadsheet.getSheetByName(RALLY_TRACKING_TRACKING_SHEET);
  if (!trackingSheet) trackingSheet = project.spreadsheet.insertSheet(RALLY_TRACKING_TRACKING_SHEET);
  rallyTrackingEnsureHeaders_(trackingSheet, RALLY_TRACKING_TRACKING_HEADERS);
  const reportKeyColumn = RALLY_TRACKING_TRACKING_HEADERS.indexOf('Report Key') + 1;
  if (reportKeyColumn > 0) {
    try { trackingSheet.hideColumns(reportKeyColumn); } catch (_) {}
  }
  rallyTrackingFormatDateColumns_(trackingSheet);

  return {
    folder: project.rallyFolder,
    spreadsheet: project.spreadsheet,
    eventsSheet: eventsSheet,
    trackingSheet: trackingSheet,
    errorsSheet: project.errorsSheet,
    rallyName: title
  };
}

function rallyTrackingEnsureHeaders_(sheet, headers) {

  if (!same) sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
  sheet.setFrozenRows(1);
}
